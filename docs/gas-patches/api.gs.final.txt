function doGet(e) {
  const result = dispatch_(e);
  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// POST エントリ（事前登録・ファイルアップロード等。
// CORSプリフライト回避のためフロントは Content-Type を付けず JSON文字列を body で送る）
function doPost(e) {
  let p = (e && e.parameter) ? e.parameter : {};
  try {
    if (e && e.postData && e.postData.contents) {
      const j = JSON.parse(e.postData.contents);
      p = Object.assign({}, p, j);
    }
  } catch (err) { /* JSONでなければ parameter のみ使用 */ }
  const result = dispatch_({ parameter: p });
  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// アクション: getStudent (企業向け / cardToken)
// ============================================================

function actionGetStudent_(p) {
  if (!p.token) return err_('missing_token');
  if (!p.event) return err_('missing_event');

  // 学生自身のMYPASS表示（source=mypass）はイベント期間中のみ許可する。
  // 企業が学生カードを確認する場合はsourceなし → publicDeadlineまで許可。
  if (p.source === 'mypass' && !isEventDay_(p.event)) return err_('event_ended');

  const cfg = getConfig_(p.event);
  if (isPastDeadline_(cfg)) return err_('expired');

  const student = findStudentByCardToken_(p.token, p.event);
  if (!student) return err_('invalid_token');

  // MYPASS からのリクエストはスタンプトークンも返す（進捗リンクに ?st= を付与するため）
  let stampToken = null;
  if (p.source === 'mypass') {
    const ps = eventSs_(p.event).getSheetByName(SHEET.STAMP_PARTICIPANTS);
    if (ps && ps.getLastRow() > 1) {
      const pd  = ps.getDataRange().getValues();
      const ph  = pd[0].map(String);
      const pEv = ph.indexOf('eventId');
      const pSi = ph.indexOf('studentId');
      const pSt = ph.indexOf('stampToken');
      const row = pd.slice(1).find(r =>
        String(r[pEv]) === String(p.event) &&
        String(r[pSi]) === String(student.studentId)
      );
      if (row && pSt >= 0) stampToken = String(row[pSt]);
    }
  }

  return ok_({
    studentId:  student.studentId,
    name:       student.name,
    furigana:   student.furigana,
    school:     student.school,
    department: student.department,
    year:       student.year,
    category:   student.category,
    clubYears:  student.clubYears,
    prefecture: student.prefecture,
    birthday:   student.birthday,
    email:      student.email,
    eventName:  cfg.eventName,
    deadline:   cfg.publicDeadline,
    stampToken: stampToken,
  });
}

// ============================================================
// アクション: saveViewLog (企業向け)
// 企業識別は2系統:
//   vk(viewKey)あり → サーバ側で企業IDに解決して自動記録 (source: company_auto)
//   companyあり     → 従来のメール手動登録等          (source: manual_email)
// VIEW_LOG 6列目に source を記録して両者を区別する。
// ============================================================

function actionSaveViewLog_(p) {
  if (!p.token) return err_('missing_token');
  if (!p.event) return err_('missing_event');

  const cfg = getConfig_(p.event);
  if (isPastDeadline_(cfg)) return err_('expired');

  const student = findStudentByCardToken_(p.token, p.event);
  if (!student) return err_('invalid_token');

  let companyId   = p.company || '';
  let companyName = '';
  let source      = 'manual_email';
  if (p.vk) {
    const company = findCompanyByViewKey_(p.vk, p.event);
    if (!company) return err_('invalid_key');
    companyId   = company.companyId;
    companyName = company.companyName;
    source      = 'company_auto';
  }
  if (!companyId) return err_('missing_params');

  const sheet = eventSs_(p.event).getSheetByName(SHEET.VIEW_LOG);

  // 自動記録の軽い重複抑制:
  // 同一イベント×同一学生×同一企業の行が既にあれば時刻のみ更新(行は増やさない)
  if (source === 'company_auto' && sheet.getLastRow() > 1) {
    const rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues();
    const idx = rows.findIndex(r =>
      String(r[1]) === String(p.event) &&
      String(r[2]) === String(student.studentId) &&
      String(r[3]) === String(companyId)
    );
    if (idx >= 0) {
      sheet.getRange(idx + 2, 1).setValue(new Date());
      return ok_({ logged: true, updated: true, companyName });
    }
  }

  sheet.appendRow([
    new Date(), p.event, student.studentId, companyId, p.email || '', source,
  ]);

  return ok_({ logged: true, companyName });
}

// アクション: resolveViewKey (企業QR登録)
// 企業QR(card.html?viewkey=...)読み取り時に viewKey の有効性を確認し、
// 企業名を返す。cookie には viewKey を保存する(企業IDはサーバ解決)。
// ============================================================

// ============================================================
// 企業閲覧用のイベント解決（2026-08-19 追加）
// ============================================================
// 企業の閲覧QRはラミネート加工して毎年使い回す運用のため、URLにイベントIDを
// 埋め込まない。代わりにサーバー側で「開始済みの最新イベント」を解決する。
//
// ⚠ getCurrentEvent（actionGetCurrentEvent_）とはルールが違うので流用しないこと。
//   getCurrentEvent は「今日が開催期間内」＝会期中しか当たらない。当日参加登録や
//   スタンプ取得は会期中しか行われないためそれで正しいが、企業の学生一覧閲覧は
//   **会期後に見るのが主用途**なので、それでは会期翌日から使えなくなる。
//
// ここでは「開始日 <= 今日」の公開中イベントのうち、開始日が最も新しいものを返す。
//   - 会期中          → そのイベント
//   - イベント間の期間 → 直前に開催したイベント（引き続き閲覧できる）
//   - 次のイベント開始 → その日から次のイベントへ自動的に切り替わる
// これにより「次のイベント開始日の前日までが前イベントの閲覧期限」となり、
// 開催間隔が短い場合でも期限設定を手で調整する必要がない。
//
// なお閲覧そのものの期限（終了日+2ヶ月 / publicDeadline）は
// 各アクションの isPastDeadline_ が別途判定する。
function resolveCompanyViewEvent_() {
  const sheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
  if (!sheet || sheet.getLastRow() < 2) return null;

  const data      = sheet.getDataRange().getValues();
  const headers   = data[0].map(String);
  const idCol     = headers.indexOf('eventId');
  const startCol  = headers.indexOf('開始日');
  const statusCol = headers.indexOf('状態');
  if (idCol < 0 || startCol < 0) return null;

  const today = new Date(); today.setHours(23, 59, 59, 999);

  let best = null, bestStart = null;
  data.slice(1).forEach(r => {
    if (statusCol >= 0) {
      const st = String(r[statusCol]);
      if (st === '公開停止' || st === '完了') return;
    }
    const start = new Date(r[startCol]);
    if (isNaN(start)) return;
    start.setHours(0, 0, 0, 0);
    if (start > today) return;                    // まだ始まっていない
    if (!bestStart || start >= bestStart) {       // 同日開始なら後の行を優先
      bestStart = start;
      best = String(r[idCol]);
    }
  });
  return best;
}

// 企業閲覧3アクション共通の入口。p.event があればそれを尊重し（配布済みの
// イベントID付きURLとの後方互換）、無ければ上のルールで解決する。
function companyViewEvent_(p) {
  if (p.event) return String(p.event);
  return resolveCompanyViewEvent_();
}

function actionResolveViewKey_(p) {
  if (!p.key)   return err_('missing_key');
  const event = companyViewEvent_(p);
  if (!event) return err_('no_active_event');

  const company = findCompanyByViewKey_(p.key, event);
  if (!company) return err_('invalid_key');

  return ok_({ companyId: company.companyId, companyName: company.companyName });
}

// アクション: activateStamp (cardToken → stampToken)
// ============================================================

function actionActivateStamp_(p) {
  if (!p.token) return err_('missing_token');
  if (!p.event) return err_('missing_event');
  if (isEventInactive_(p.event)) return err_('event_inactive');

  const cfg = getConfig_(p.event);
  if (!isStampOpen_(cfg)) return err_('stamp_closed');

  const student = findStudentByCardToken_(p.token, p.event);
  if (!student) return err_('invalid_token');

  const lock = LockService.getScriptLock();
  try { lock.waitLock(6000); } catch (e) { return err_('lock_timeout'); }
  try {
    const ss_ = eventSs_(p.event);
    let participantSheet = ss_.getSheetByName(SHEET.STAMP_PARTICIPANTS);
    if (!participantSheet) {
      participantSheet = ss_.insertSheet(SHEET.STAMP_PARTICIPANTS);
      participantSheet.appendRow(['時刻', 'eventId', 'studentId', 'stampToken', 'activatedAt', 'status']);
    }

    let stampToken = null;
    if (participantSheet.getLastRow() > 1) {
      const data    = participantSheet.getDataRange().getValues();
      const headers = data[0].map(String);
      const evCol   = headers.indexOf('eventId');
      const sidCol  = headers.indexOf('studentId');
      const stCol   = headers.indexOf('stampToken');
      const existing = data.slice(1).find(r =>
        String(r[evCol]) === String(p.event) &&
        String(r[sidCol]) === String(student.studentId)
      );
      if (existing) stampToken = String(existing[stCol]);
    }

    let isNew = false;
    if (!stampToken) {
      stampToken = Utilities.getUuid();
      isNew = true;
      participantSheet.appendRow([
        new Date(), p.event, student.studentId,
        stampToken, new Date(), 'active',
      ]);
    }

    const count = countStamps_(p.event, student.studentId);
    return ok_(buildStampResult_(p.event, student.studentId, count, { stampToken, isNew }));
  } finally {
    lock.releaseLock();
  }
}

// ============================================================
// アクション: saveStamp (stampToken)
// ============================================================

function actionSaveStamp_(p) {
  if (!p.st || !p.ct) return err_('missing_params');
  if (!p.event) return err_('missing_event');
  if (isEventInactive_(p.event)) return err_('event_inactive');
  const cfg = getConfig_(p.event);
  if (!isStampOpen_(cfg)) return err_('stamp_closed');

  const student = findStudentByStampToken_(p.st, p.event);
  if (!student) return err_('invalid_student_token');

  const company = findCompanyByStampKey_(p.ct, p.event);
  if (!company) return err_('invalid_company_key');
  // スタンプ帳の表示からは除外していても、既設のNFCタグからは加算できてしまっていた。
  // 非表示のブースのスタンプが景品獲得数に効いてしまうため、ここでも弾く。
  if (!company.stampRally) return err_('company_not_in_rally');

  const lock = LockService.getScriptLock();
  try { lock.waitLock(6000); } catch (e) { return err_('lock_timeout'); }
  try {
    const sheet = eventSs_(p.event).getSheetByName(SHEET.STAMP_LOG);
    const rows  = sheet.getLastRow() > 1
      ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).getValues()
      : [];

    // 同じブースでも「日が変われば」再度取得できる（日別リセット）
    const dup = rows.some(r =>
      String(r[1]) === String(p.event) &&
      String(r[2]) === String(student.studentId) &&
      String(r[7]) === String(company.companyId) &&
      sameDayJst_(r[0])
    );
    if (dup) return err_('already_stamped');

    if (p.nc) {
      const ncUsed = rows.some(r =>
        String(r[7]) === String(company.companyId) &&
        String(r[8]) === String(p.nc)
      );
      if (ncUsed) return err_('used_nfc');
    }

    sheet.appendRow([
      new Date(), p.event, student.studentId,
      student.name, student.school, student.category,
      company.companyName, company.companyId, p.nc || '',
    ]);

    const prevCount = rows.filter(r =>
      String(r[1]) === String(p.event) &&
      String(r[2]) === String(student.studentId) &&
      sameDayJst_(r[0])
    ).length;
    const count = prevCount + 1;

    // 今回のスタンプで景品条件を達成したか (cfg は上で取得済み)
    // ⚠ キーの優先順は computePrizeState_ と必ず揃えること。
    //   新規イベントは setupEventSpreadsheet_ が prizeUnitSize を書き、管理画面も
    //   prizeUnitSize を保存するため、prizeThreshold を先に見ると通常は存在せず
    //   ハードコードの 5 にフォールバックしてしまう。
    const threshold = Number(cfg.prizeUnitSize) || Number(cfg.prizeThreshold) || 5;
    const justCleared = prevCount < threshold && count >= threshold;

    return ok_(buildStampResult_(p.event, student.studentId, count, {
      company: company.companyName,
      justCleared,
    }));
  } finally {
    lock.releaseLock();
  }
}

// アクション: getStampProgress (stampToken)
// ============================================================

function actionGetStampProgress_(p) {
  if (!p.token && !p.pk) return err_('missing_token');
  if (!p.event) return err_('missing_event');

  if (!isEventDay_(p.event)) return err_('event_ended');

  // stampToken で学生を特定。未登録(STAMP_PARTICIPANTS未作成)の場合は cardToken(pk)で fallback。
  let student = p.token ? findStudentByStampToken_(p.token, p.event) : null;
  if (!student && p.pk) student = findStudentByCardToken_(p.pk, p.event);
  if (!student) return err_('invalid_token');

  const sheet = eventSs_(p.event).getSheetByName(SHEET.STAMP_LOG);
  const rows  = sheet.getLastRow() > 1
    ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).getValues()
    : [];

  const myStamps = rows
    .filter(r =>
      String(r[1]) === String(p.event) &&
      String(r[2]) === String(student.studentId) &&
      sameDayJst_(r[0]))
    .map(r => ({
      company:   r[6],
      companyId: String(r[7] || ''),   // スタンプ帳グリッドのID突合用
      time: r[0] instanceof Date
        ? Utilities.formatDate(r[0], Session.getScriptTimeZone(), 'yyyy/MM/dd HH:mm')
        : String(r[0]),
    }));

  return ok_(buildStampResult_(p.event, student.studentId, myStamps.length, {
    stamps: myStamps,
    companies: getCompaniesForEvent_(p.event),  // スタンプ帳の全ブース（ロゴ含む）
    name:      student.name,        // 本人確認用（共有端末での取り違え防止）
    studentId: student.studentId,   // 同上・補助識別
  }));
}

// イベントの全企業（スタンプ帳グリッド用・表示順）
// stampRally が明示的に 'FALSE' の企業はグリッドから除外する
function getCompaniesForEvent_(eventId) {
  const sheet = eventSs_(eventId).getSheetByName(SHEET.COMPANIES);
  if (!sheet || sheet.getLastRow() < 2) return [];
  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const g       = (row, name) => { const i = headers.indexOf(name); return i >= 0 ? row[i] : ''; };
  const orderCol = headers.indexOf('表示順');
  const rows = data.slice(1)
    .filter(row => String(g(row, 'stampRally')).toUpperCase() !== 'FALSE')
    .map(row => ({
      companyId: String(g(row, 'companyId')),
      name:      String(g(row, '企業名')),
      logoUrl:   String(g(row, 'logoUrl') || ''),
      _order:    orderCol >= 0 ? Number(row[orderCol]) || 0 : 0,
    }));
  rows.sort((a, b) => a._order - b._order);
  return rows.map(({ _order, ...c }) => c);
}

// アクション: exchangePrize (学生側確定 / stampToken)        ← T3-d 新規実装
// 学生が progress.html で景品交換を確定する。1人1回のみ。
// ============================================================

function actionExchangePrize_(p) {
  if (!p.token && !p.pk) return err_('missing_token');
  if (!p.event) return err_('missing_event');
  if (isEventInactive_(p.event)) return err_('event_inactive');
  const cfg = getConfig_(p.event);
  if (!isExchangeOpen_(cfg)) return err_('exchange_closed');
  // stampToken で学生特定。未登録なら cardToken(pk)で fallback（学生確定→studentIdで記録）
  let student = p.token ? findStudentByStampToken_(p.token, p.event) : null;
  if (!student && p.pk) student = findStudentByCardToken_(p.pk, p.event);
  if (!student) return err_('invalid_token');

  // 同時リクエストによる二重交換を防ぐためロック内で再計算してから書き込む
  // ⚠ waitLock を素で呼ばない理由は actionRegisterWalkIn_ 側のコメントを参照。
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (e) { return err_('lock_timeout'); }
  try {
    const count  = countStamps_(p.event, student.studentId);
    const result = buildStampResult_(p.event, student.studentId, count, {});
    if (result.claimableNow <= 0) return err_('nothing_to_claim');
    eventSs_(p.event).getSheetByName(SHEET.PRIZE_LOG).appendRow([
      new Date(), p.event, student.studentId, student.name,
      count, '学生確定', result.claimableNow,
    ]);
    // appendRow後に countExchanged_ が更新されるので再計算して返す
    return ok_(buildStampResult_(p.event, student.studentId, count, {}));
  } finally {
    lock.releaseLock();
  }
}

// アクション: getSchoolList（大学マスターの名前一覧を返す）
// ============================================================

function actionGetSchoolList_(p) {
  // コード順に並べる。コード先頭2桁は読み頭文字のポケベル値（行→段）なので、
  // コード昇順 ≒ 五十音順（頭文字基準）。漢字名でも正しい五十音順で並ぶ。
  // ただし わ/を/ん は 01/05/00、仮コードは 0001〜 と先頭桁が 0 で、そのままだと
  // 先頭に来てしまう。先頭桁 '0' を '9' の後ろ（':'）に送るキーで末尾側へ補正する。
  const sortKey = c => {
    c = String(c).padStart(4, '0');
    return (c.charAt(0) === '0' ? ':' : c.charAt(0)) + c.substring(1);
  };
  const { uniMap } = loadUniversityMaster_();
  const schools = Array.from(uniMap.entries())
    .sort((a, b) => {
      const ka = sortKey(a[1]), kb = sortKey(b[1]);
      return ka < kb ? -1 : ka > kb ? 1 : 0; // localeCompareは':'を特殊扱いするためコードポイント比較
    })
    .map(e => e[0]);
  return ok_({ schools });
}

/**
 * 学生が景品交換した累計景品数
 * PRIZE_LOG の claimedCount 列を合計する。
 * 旧形式（claimedCount列なし）は行数をカウント（1行=1個扱い）。
 */
function countExchanged_(eventId, studentId) {
  const sheet = eventSs_(eventId).getSheetByName(SHEET.PRIZE_LOG);
  if (!sheet || sheet.getLastRow() < 2) return 0;

  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const evCol   = headers.indexOf('eventId');
  const sidCol  = headers.indexOf('studentId');
  const cntCol  = headers.indexOf('claimedCount');
  const timeCol = headers.indexOf('時刻');

  // 景品も日別にリセットする（土曜に上限まで交換した学生も、日曜は改めて交換できる）。
  // 「より多くの企業を回ってもらう」ための運用判断（2026-08-22）。
  return data.slice(1)
    .filter(r => String(r[evCol]) === String(eventId) &&
                 String(r[sidCol]) === String(studentId) &&
                 (timeCol < 0 || sameDayJst_(r[timeCol])))
    .reduce((sum, r) => sum + (cntCol >= 0 ? Number(r[cntCol]) || 1 : 1), 0);
}

/** スタンプ数を数える */
// ============================================================
// スタンプ・景品の「当日分のみ」集計（2026-08-22 追加）
// ============================================================
// 土日2日間のイベントで、日曜は0からスタンプラリーを開始し直す運用のため、
// 集計を暦日単位に切り替える。
//
// ⚠ データは一切書き換えない。STAMP_LOG / PRIZE_LOG の行はそのまま残し、
//   「数えるときに当日分だけを対象にする」ことでリセットを実現する。
//   これにより、土曜の記録を失わずに日曜を0から始められる。
//
// ⚠ 企業の来訪学生一覧（actionGetCompanyStampVisitors_）はこの絞り込みを
//   **使わない**。企業には土日をまとめて見せる必要があるため、
//   あちらは eventId × companyId のままにしてある。変更しないこと。
//
// 競技時間は9:00〜17:00で日付をまたがないため、境界は暦日で判定して問題ない。
function sameDayJst_(v, ref) {
  if (!(v instanceof Date)) {
    v = new Date(v);
    if (isNaN(v)) return false;
  }
  const tz  = Session.getScriptTimeZone();
  const fmt = d => Utilities.formatDate(d, tz, 'yyyy/MM/dd');
  return fmt(v) === fmt(ref || new Date());
}

function countStamps_(eventId, studentId) {
  const sheet = eventSs_(eventId).getSheetByName(SHEET.STAMP_LOG);
  const rows  = sheet.getLastRow() > 1
    ? sheet.getRange(2, 1, sheet.getLastRow() - 1, 9).getValues()
    : [];
  // 当日分のみ（日別リセット。sameDayJst_ のコメント参照）
  return rows.filter(r =>
    String(r[1]) === String(eventId) &&
    String(r[2]) === String(studentId) &&
    sameDayJst_(r[0])
  ).length;
}

/**
 * スタンプ結果の共通レスポンスを組み立てる（累積交換モデル）
 *
 * 景品ロジック:
 *   eligibleTotal = min( floor(stampCount / prizeUnitSize), maxPrizes )
 *   claimableNow  = max( 0, eligibleTotal - exchangedCount )
 *   cleared       = claimableNow > 0
 *   exchanged     = exchangedCount >= maxPrizes（これ以上交換不可）
 */
function buildStampResult_(eventId, studentId, stampCount, extra) {
  const cfg           = getConfig_(eventId);
  const prizeUnitSize = Number(cfg.prizeUnitSize) || Number(cfg.prizeThreshold) || 5;
  const maxPrizes     = Number(cfg.maxPrizes)     || Number(cfg.prizeCount)     || 3;
  const exchangedCount  = countExchanged_(eventId, studentId);
  const eligibleTotal   = Math.min(Math.floor(stampCount / prizeUnitSize), maxPrizes);
  const claimableNow    = Math.max(0, eligibleTotal - exchangedCount);
  // 次の交換閾値（これ以上交換不可なら null）
  const nextTier        = exchangedCount + 1;
  const nextThreshold   = nextTier <= maxPrizes ? nextTier * prizeUnitSize : null;

  return Object.assign({
    stampCount,
    prizeUnitSize,
    maxPrizes,
    eligibleTotal,
    exchangedCount,
    claimableNow,
    nextThreshold,
    cleared:  claimableNow > 0,
    exchanged: exchangedCount >= maxPrizes,
  }, extra || {});
}

// スプレッドシートアクセス
// ============================================================
//
// SpreadsheetApp.openById は GAS で最も重い操作の一つ。1リクエストで eventSs_ が
// 4〜5回呼ばれ、毎回マスター＋イベントの2回 openById ＋ EVENT_LIST 全読みが走る。
// doGet/doPost ごとに新しいJSコンテキストになるため、モジュール変数でリクエスト単位に
// メモ化しても他リクエストと衝突しない（同一インスタンス再利用時も Spreadsheet は
// ライブ参照なので最新データを返し整合性は保たれる）。

var SS_CACHE_      = null;   // master Spreadsheet をメモ化
var EVENT_SS_CACHE_ = {};    // eventId → Spreadsheet をメモ化

function masterSs_() {
  if (SS_CACHE_) return SS_CACHE_;
  const id = PropertiesService.getScriptProperties()
    .getProperty('MASTER_SPREADSHEET_ID');
  if (!id) throw new Error('MASTER_SPREADSHEET_IDが未設定です。initializeMaster()を実行してください。');
  SS_CACHE_ = SpreadsheetApp.openById(id);
  return SS_CACHE_;
}

function eventSs_(eventId) {
  const key = String(eventId);
  if (EVENT_SS_CACHE_[key]) return EVENT_SS_CACHE_[key];
  const sheet   = masterSs_().getSheetByName(SHEET.EVENT_LIST);
  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const idCol   = headers.indexOf('eventId');
  const ssIdCol = headers.indexOf('スプレッドシートID');
  const row     = data.slice(1).find(r => String(r[idCol]) === String(eventId));
  if (!row) throw new Error('イベントが見つかりません: ' + eventId);
  const ss = SpreadsheetApp.openById(row[ssIdCol]);
  EVENT_SS_CACHE_[key] = ss;
  return ss;
}

function getConfig_(eventId) {
  const cache = CacheService.getScriptCache();
  const key   = 'cfg_' + eventId;
  const hit   = cache.get(key);
  if (hit) return JSON.parse(hit);

  const data   = eventSs_(eventId).getSheetByName(SHEET.CONFIG).getDataRange().getValues();
  const config = {};
  data.forEach(([k, v]) => { if (k) config[String(k)] = v; });
  cache.put(key, JSON.stringify(config), 300);
  return config;
}

// 企業の来訪学生一覧などの閲覧期限。
// 既定は「イベント終了日+2ヶ月」を下限とし、管理者がそれより後の publicDeadline を
// 明示設定した場合のみ延長として採用する（短縮は不可＝最低2ヶ月は閲覧可）。
function isPastDeadline_(cfg) {
  var deadline = null;
  if (cfg.endDate) {
    var end = new Date(cfg.endDate);
    if (!isNaN(end)) {
      deadline = new Date(end);
      deadline.setMonth(deadline.getMonth() + 2);
      deadline.setHours(23, 59, 59);
    }
  }
  if (cfg.publicDeadline) {
    var pd = new Date(cfg.publicDeadline);
    if (!isNaN(pd) && (!deadline || pd > deadline)) deadline = pd; // 2ヶ月後より後なら延長
  }
  if (!deadline) return false; // 終了日も期限も不明なら無期限
  return new Date() > deadline;
}

  const startCol = headers.indexOf('開始日');
  const endCol   = headers.indexOf('終了日');
  const row = data.slice(1).find(r => String(r[idCol]) === String(eventId));
  if (!row) return false;
  const today = new Date(); today.setHours(0, 0, 0, 0);
  const start = new Date(row[startCol]); start.setHours(0, 0, 0, 0);
  const end   = new Date(row[endCol]);   end.setHours(23, 59, 59, 999);
  return today >= start && today <= end;
}

function findStudent_(keyCol, value, eventId) {

function mapStudent_(headers, row) {
  const g = name => { const i = headers.indexOf(name); return i >= 0 ? row[i] : ''; };
  return {
    studentId:  g('studentId'),
    cardToken:  g('cardToken'),   // T11: token → cardToken
    name:       g('氏名'),
    furigana:   g('ふりがな'),
    school:     g('大学名'),
    department: g('学部学科'),
    year:       g('学年'),
    category:   g('属性'),
    clubYears:  g('自動車部在籍年数'),
    prefecture: g('住所(都道府県)'),
    birthday:   g('生年月日') instanceof Date
      ? Utilities.formatDate(g('生年月日'), Session.getScriptTimeZone(), 'yyyy/MM/dd')
      : String(g('生年月日') || ''),
    email:      g('メールアドレス'),
    regType:    g('登録種別'),
  };
}

// STUDENTSシートを1回だけ読み込み、studentId→mapStudent_と同じ形の学生情報を持つ
// Mapを作る。企業向け一覧（閲覧履歴・スタンプ来訪者）のように、複数行それぞれで
// findStudentById_を呼ぶとシート全体を毎回読み直すN+1になるため、そのループの前に
// 1回だけ呼んで使い回す用途の共有ヘルパー。mapStudent_をそのまま使うため、
// 個別にfindStudentById_した場合と返る形は完全に同一。
function buildStudentIndex_(eventId) {
  const map = new Map();
  const sheet = eventSs_(eventId).getSheetByName(SHEET.STUDENTS);
  if (!sheet || sheet.getLastRow() < 2) return map;
  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const idCol   = headers.indexOf('studentId');
  if (idCol < 0) return map;
  data.slice(1).forEach(row => {
    const sid = String(row[idCol] || '');
    if (sid) map.set(sid, mapStudent_(headers, row));
  });
  return map;
}

// アクション: getCurrentEvent (T6)
// 今日の日付に合致するアクティブなイベントを返す
// ============================================================

function actionGetCurrentEvent_(p) {
  const sheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
  if (!sheet || sheet.getLastRow() < 2) return err_('no_active_event');

  const data      = sheet.getDataRange().getValues();
  const headers   = data[0].map(String);
  const idCol     = headers.indexOf('eventId');
  const nameCol   = headers.indexOf('イベント名');
  const startCol  = headers.indexOf('開始日');
  const endCol    = headers.indexOf('終了日');
  const statusCol = headers.indexOf('状態');

  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const row = data.slice(1).find(r => {
    // 「公開停止」(旧データ互換で「完了」も)は開催日内でも公開対象から除外する
    if (statusCol >= 0) {
      const st = String(r[statusCol]);
      if (st === '公開停止' || st === '完了') return false;
    }
    const start = new Date(r[startCol]); start.setHours(0, 0, 0, 0);
    const end   = new Date(r[endCol]);   end.setHours(23, 59, 59, 999);
    return today >= start && today <= end;
  });

  if (!row) return err_('no_active_event');

  const eventId = String(row[idCol]);
  // 書類URL（当日フォームなどが同期取得できるように含める）
  let docUrls = { rulebook: '', pledge: '', schoolApproval: '' };
  try {
    const cfg = getConfig_(eventId);
    docUrls = {
      rulebook:       String(cfg.docUrlRulebook       || ''),
      pledge:         String(cfg.docUrlPledge         || ''),
      schoolApproval: String(cfg.schoolEntryApprovalUrl || ''),
    };
  } catch (e) { /* 失敗時は空URL（HTMLフォールバックが使われる） */ }

  return ok_({
    eventId:   eventId,
    eventName: row[nameCol],
    startDate: row[startCol] instanceof Date
      ? Utilities.formatDate(row[startCol], Session.getScriptTimeZone(), 'yyyy/MM/dd')
      : String(row[startCol]),
    endDate: row[endCol] instanceof Date
      ? Utilities.formatDate(row[endCol], Session.getScriptTimeZone(), 'yyyy/MM/dd')
      : String(row[endCol]),
    docUrls: docUrls,
  });
}

// ============================================================
// アクション: registerWalkIn (T6)
// 当日飛び込み参加者を登録してstampTokenを発行する。
// 同日・同一学生(氏名+大学名)の重複登録は既存stampTokenを返す。
// ============================================================

function actionRegisterWalkIn_(p) {
  if (!p.event) return err_('missing_event');
  if (!p.name || !p.school) return err_('missing_params');

  // サーバー側検証: イベント状態・開催日チェック
  if (isEventInactive_(p.event)) return err_('event_inactive');
  if (!isEventDay_(p.event))     return err_('not_event_day');

  const studentSheet = eventSs_(p.event).getSheetByName(SHEET.STUDENTS);

  // 重複チェック → 採番 → appendRow を同一ロック内で完結させ、同時登録での ID 重複を防ぐ
  // ⚠ waitLock を素で呼ぶと、取得できなかったとき例外が dispatch_ の catch に落ちて
  //   err_('server_error') になる。大学確定・統合はスイープ完了までロックを保持するため
  //   （意図的な設計）、その最中の登録がここに来る。「待てば通る」状況なのに
  //   「サーバーエラー」と出ると学生が諦めてしまうので、他のロック取得箇所と同じく
  //   lock_timeout を返して再試行を促す。
  const lock = LockService.getScriptLock();
  try { lock.waitLock(15000); } catch (e) { return err_('lock_timeout'); }
  let outcome_; // ロック内で確定した結果を外へ持ち出す
  try {
    const sData   = studentSheet.getDataRange().getValues();
    const sHeaders = sData[0].map(String);
    const nameCol  = sHeaders.indexOf('氏名');
    const schoolCol = sHeaders.indexOf('大学名');
    const sidCol   = sHeaders.indexOf('studentId');

    // 同日・同一学生の重複チェック(氏名+大学名)
    // 空白の表記揺れ（半角/全角/前後）で別人扱いにならないよう正規化して照合する。
    // （クッキー喪失→再登録での復帰の取りこぼしを防ぐ）
    const norm_ = s => String(s == null ? '' : s).replace(/[\s　]/g, '');
    const existing = sData.slice(1).find(r =>
      norm_(r[nameCol])   === norm_(p.name) &&
      norm_(r[schoolCol]) === norm_(p.school)
    );

    if (existing) {
      const existStudentId = String(existing[sidCol]);
      let participantSheet = eventSs_(p.event).getSheetByName(SHEET.STAMP_PARTICIPANTS);
      if (!participantSheet) {
        participantSheet = eventSs_(p.event).insertSheet(SHEET.STAMP_PARTICIPANTS);
        participantSheet.appendRow(['時刻', 'eventId', 'studentId', 'stampToken', 'activatedAt', 'status']);
      }
      const pData    = participantSheet.getDataRange().getValues();
      const pHeaders = pData[0].map(String);
      const pSidCol  = pHeaders.indexOf('studentId');
      const pStCol   = pHeaders.indexOf('stampToken');
      const pRow     = pData.slice(1).find(r =>
        String(r[pSidCol]) === existStudentId &&
        String(r[pHeaders.indexOf('eventId')]) === String(p.event)
      );
      // 既存学生だが参加者行が無い場合（例: 事前登録済みで未開始）は
      // 新規トークンを発行し、必ずシートへ追記する。
      // （追記しないと未保存トークンを返し、以後スタンプ不能になるため）
      let stampToken;
      if (pRow) {
        stampToken = String(pRow[pStCol]);
      } else {
        stampToken = Utilities.getUuid();
        participantSheet.appendRow([
          new Date(), p.event, existStudentId, stampToken, new Date(), 'active',
        ]);
      }
      const existCardToken = String(existing[sHeaders.indexOf('cardToken')] || '');
      const count = countStamps_(p.event, existStudentId);
      const result = buildStampResult_(p.event, existStudentId, count, { stampToken, cardToken: existCardToken });
      outcome_ = { type: 'existing', existStudentId, existCardToken, stampResult: result };
    } else {
      // メアド重複チェック（部活代表や共有アドレスの使い回し抑止）
      // フロントで警告表示→同意チェックを経由した場合のみ emailDupAcknowledged=true が送られてくる。
      const acknowledged = String(p.emailDupAcknowledged) === 'true';
      if (!acknowledged && p.email) {
        const emailCol = sHeaders.indexOf('メールアドレス');
        const target = String(p.email).trim().toLowerCase();
        if (emailCol >= 0 && target) {
          const dup = sData.slice(1).some(r => String(r[emailCol]).trim().toLowerCase() === target);
          if (dup) return err_('email_already_used');
        }
      }
      // 新規: studentId 採番・両シートへの appendRow をロック内で完結
      const yearMatch  = String(p.event).match(/(\d{4})/);
      const yearSuffix = yearMatch ? yearMatch[1].slice(-1) : String(new Date().getFullYear()).slice(-1);
      const { uniMap, maxAutoCode } = loadUniversityMaster_();
      const schoolName = String(p.school).trim();
      let uniCode = uniMap.get(schoolName);
      if (!uniCode) {
        uniCode = String(maxAutoCode + 1).padStart(4, '0');
        saveNewUniversities_([[schoolName, uniCode, '自動割当(要確認)']]);
      }
      // 連番は「西暦+大学コード+区分文字(=V)」で始まる既存IDの最大連番+1（＝同一大学・同一区分のみ）。
      // 全件通算ではなく大学・区分単位で 01 から採番。max+1 なので既存IDの歯抜けとも衝突しない。
      const idPrefix  = yearSuffix + uniCode + 'V';
      const studentId = idPrefix + String(nextStudentSeq_(sData, sidCol, idPrefix)).padStart(2, '0');
      const token     = Utilities.getUuid();

      // 学生マスターに追加
      const newRow = sHeaders.map(h => {
        const map = {
          'studentId': studentId, 'cardToken': token,  // T11: token → cardToken
          '氏名': String(p.name).trim(), 'ふりがな': String(p.furigana || '').trim(),
          '大学名': String(p.school).trim(), '学部学科': p.department || '',
          '学年': p.year || '', '属性': '一般参加学生',
          '自動車部在籍年数': p.clubYears || '',
          '住所(都道府県)': p.prefecture || '',
          '性別': p.gender || '',                       // フォーム準拠で追加
          '生年月日': p.birthday || '', 'メールアドレス': p.email || '',
          '電話番号': p.phone || '',
          '所属校出場': p.competing || '',              // 所属校が出場するか
          '規則誓約同意': p.rulesConsent  === 'true' ? 'true' : 'false', // 同意記録(大会規則書・誓約書)
          'SNS同意':     p.snsConsent     === 'true' ? 'true' : 'false', // 同意記録
          '個人情報同意': p.privacyConsent === 'true' ? 'true' : 'false', // 同意記録
          '登録種別': '当日',
        };
        return map[h] !== undefined ? map[h] : '';
      });
      const phoneCol = sHeaders.indexOf('電話番号');
      const nextStudentRow = studentSheet.getLastRow() + 1;
      if (phoneCol >= 0) {
        studentSheet.getRange(nextStudentRow, phoneCol + 1).setNumberFormat('@');
      }
      studentSheet.getRange(nextStudentRow, 1, 1, newRow.length).setValues([newRow]);

      // スタンプ参加者シートに追加（シート未存在時は自動作成）
      const stampToken = Utilities.getUuid();
      let stampParticipantSheet = eventSs_(p.event).getSheetByName(SHEET.STAMP_PARTICIPANTS);
      if (!stampParticipantSheet) {
        stampParticipantSheet = eventSs_(p.event).insertSheet(SHEET.STAMP_PARTICIPANTS);
        stampParticipantSheet.appendRow(['時刻', 'eventId', 'studentId', 'stampToken', 'activatedAt', 'status']);
      }
      stampParticipantSheet.appendRow([
        new Date(), p.event, studentId, stampToken, new Date(), 'active',
      ]);
      outcome_ = { type: 'new', studentId, token, stampToken };
    }
  } finally {
    lock.releaseLock();
  }

  // ロック解放後にメール送信（IO待ちでロックを保持しない）
  if (outcome_.type === 'existing') {
    const { existStudentId, existCardToken, stampResult } = outcome_;
    // 再登録は実質「復帰」操作。個人ページ(氏名+QR)のリンクを再送（失敗しても登録判定は維持）
    try {
      sendWalkInPassMail_(p.email, p.name, p.event, buildPassUrl_(p.appBase, existCardToken, p.event));
    } catch (err) {
      console.error('sendWalkInPassMail_ (already_registered) failed:', err);
    }
    return { ok: false, error: 'already_registered', data: stampResult,
             message: 'すでに本日のイベントに登録済みです' };
  } else {
    const { studentId, token, stampToken } = outcome_;
    // 個人ページ(氏名+QR)のリンクをメール送信（失敗しても登録は成立させる）
    try {
      sendWalkInPassMail_(p.email, p.name, p.event, buildPassUrl_(p.appBase, token, p.event));
    } catch (err) {
      console.error('sendWalkInPassMail_ failed:', err);
    }
    return ok_(buildStampResult_(p.event, studentId, 0, { stampToken, isNew: true, cardToken: token }));
  }
}

// アクション: registerPreStudent (事前登録 / 脱Googleフォーム)
// 会期前に学生を事前登録する。公開（codeなし）。
// - STUDENTS に本人確認サブセットを upsert（登録種別=事前・cardToken発行）
// - 「事前登録」シートにフル記録（区分別項目・ファイルURLは後フェーズ）
// - 確認メールを送信（QR名刺は載せない）

// ============================================================

function actionRegisterPreStudent_(p) {
  if (!p.event) return err_('missing_event');
  if (!p.name || !p.school || !p.email) return err_('missing_params');

  // サーバー側検証: イベント状態・受付期間・区分別締切
  if (isEventInactive_(p.event))  return err_('event_inactive');
  const cfg  = getConfig_(p.event);
  if (isBeforeFormOpen_(cfg))     return err_('form_not_open');
  // 参加区分を先行確定して締切チェックに使う（後段でも同じ値を流用）
  const attr = preAttr_(p.category, p.driverClass, p.serviceClass);
  if (isPastCategoryDeadline_(cfg, attr)) return err_('registration_closed');

  const ss           = eventSs_(p.event);
  const studentSheet = ss.getSheetByName(SHEET.STUDENTS);

  // 重複チェック → 採番 → STUDENTS appendRow を同一ロック内で完結させ、同時登録での ID 重複を防ぐ
  // ⚠ waitLock を素で呼ばない理由は actionRegisterWalkIn_ 側のコメントを参照。
  const lock = LockService.getScriptLock();
  try { lock.waitLock(15000); } catch (e) { return err_('lock_timeout'); }
  let studentId, cardToken;
  try {
    const sData   = studentSheet.getDataRange().getValues();
    const sHeaders = sData[0].map(String);
    const idx     = n => sHeaders.indexOf(n);

    // 重複（氏名+大学名+登録種別=事前）→ 既存 cardToken を返す
    const existing = sData.slice(1).find(r =>
      String(r[idx('氏名')])     === String(p.name).trim() &&
      String(r[idx('大学名')])   === String(p.school).trim() &&
      String(r[idx('登録種別')]) === '事前'
    );
    if (existing) {
      return { ok: false, error: 'already_registered',
        data: { studentId: String(existing[idx('studentId')]), cardToken: String(existing[idx('cardToken')]) },
        message: 'すでに事前登録済みです' };
    }

    // メアド重複チェック（部活代表や共有アドレスの使い回し抑止）
    // フロントで警告表示→同意チェックを経由した場合のみ emailDupAcknowledged=true が送られてくる。
    const acknowledged = String(p.emailDupAcknowledged) === 'true';
    if (!acknowledged) {
      const emailCol = idx('メールアドレス');
      const target = String(p.email).trim().toLowerCase();
      if (emailCol >= 0 && target) {
        const dup = sData.slice(1).some(r => String(r[emailCol]).trim().toLowerCase() === target);
        if (dup) return err_('email_already_used');
      }
    }

    // 参加区分 → クラスコード（attr は上で採番前に確定済み）
    const classCode = PRE_CLASS_CODES[attr] || 'V';

    // studentId 生成
    const yearMatch  = String(p.event).match(/(\d{4})/);
    const yearSuffix = yearMatch ? yearMatch[1].slice(-1) : String(new Date().getFullYear()).slice(-1);
    const { uniMap, maxAutoCode } = loadUniversityMaster_();
    const schoolName = String(p.school).trim();
    let uniCode = uniMap.get(schoolName);
    if (!uniCode) {
      uniCode = String(maxAutoCode + 1).padStart(4, '0');
      saveNewUniversities_([[schoolName, uniCode, '自動割当(要確認)']]);
    }
    // 連番は「西暦+大学コード+区分文字」で始まる既存IDの最大連番+1。
    // 属性ではなく studentId 接頭辞で数えるため、同一文字に集約される区分
    // （補欠/メカニック/応援＝S）どうしの衝突も防げる。max+1 で歯抜けとも衝突しない。
    const idPrefix = yearSuffix + uniCode + classCode;
    studentId      = idPrefix + String(nextStudentSeq_(sData, idx('studentId'), idPrefix)).padStart(2, '0');
    cardToken      = Utilities.getUuid();

    // STUDENTS upsert（本人確認サブセット）
    const newRow = sHeaders.map(h => {
      const map = {
        'studentId': studentId, 'cardToken': cardToken,
        '氏名': String(p.name).trim(), 'ふりがな': String(p.furigana || '').trim(),
        '大学名': String(p.school).trim(), '学部学科': p.department || '',
        '学年': p.year || '', '属性': attr, '自動車部在籍年数': p.clubYears || '',
        '住所(都道府県)': p.prefecture || '', '生年月日': p.birthday || '',
        '性別': p.gender || '', 'メールアドレス': p.email || '', '電話番号': p.phone || '',
        '所属校出場': p.competing || '',
        '規則誓約同意': p.rulesConsent   === 'true' ? 'true' : 'false',
        'SNS同意':     p.mediaConsent    === 'true' ? 'true' : 'false',
        '個人情報同意': p.privacyConsent  === 'true' ? 'true' : 'false',
        '登録種別': '事前',
      };
      return map[h] !== undefined ? map[h] : '';
    });
    const phoneIdx = idx('電話番号');
    const nextPreRow = studentSheet.getLastRow() + 1;
    if (phoneIdx >= 0) {
      studentSheet.getRange(nextPreRow, phoneIdx + 1).setNumberFormat('@');
    }
    studentSheet.getRange(nextPreRow, 1, 1, newRow.length).setValues([newRow]);

    // ⚠ 事前登録シートも必ず同じロック内で書く（P0対策）。
    //   ここをロック外に出すと、学生マスター書込→事前登録書込 の隙間に
    //   大学確定・大学統合が走った場合、同じ申込なのに
    //   学生マスターは新ID・事前登録は旧ID という分裂が起きる。
    //   ファイルURLはこの時点では未確定なので空で書き、Drive保存後に該当セルだけ更新する
    //   （アップロード中もロックを保持すると、締切直前の同時登録が waitLock で詰まるため）。
    p.pledgeUrl = ''; p.licenseUrl = ''; p.insuranceUrl = ''; p.approvalUrl = '';
    appendPreRegRow_(ss, p, studentId, cardToken);
  } finally {
    lock.releaseLock();
  }

  // ファイル（Drive保存・非公開）
  // 権限未承認/一時障害でも登録自体は成立させる（サーバーエラーにしない）が、
  // 「保存できなかった」ことは filesSaved:false として必ず呼び出し元へ伝える。
  let fileUrls = {};
  let filesSaved = true;
  try {
    fileUrls = savePreRegFiles_(p.event, studentId, p.files, String(p.name || '').trim());
  } catch (err) {
    filesSaved = false;
    console.error('savePreRegFiles_ failed:', err);
  }
  // 送られてきたファイルのうち1つでもURLが得られていなければ失敗として扱う
  // （savePreRegFiles_ はファイル単位で例外を握りつぶすため、ここで突き合わせる）
  if (!expectedFilesAllSaved_(p.files, fileUrls)) filesSaved = false;

  // 保存できたURLだけ、事前登録シートの該当セルへ後追いで書き込む
  try {
    updatePreRegFileUrls_(ss, studentId, fileUrls);
  } catch (err) {
    filesSaved = false;
    console.error('updatePreRegFileUrls_ failed:', err);
  }

  // 確認メール（QR名刺は載せない・失敗しても登録は成立）
  // ⚠ sendPreRegMail_ は例外を投げず true/false を返す（2026-08-25 変更）。
  //   以前はここで try/catch しか見ておらず、関数内部で握り潰された失敗を
  //   検知できず mailSent が常に true になっていた。
  let mailSent = false;
  try {
    mailSent = sendPreRegMail_(p.email, p.name, p.event);
  } catch (err) {
    mailSent = false;
    console.error('sendPreRegMail_ failed:', err);
  }

  return ok_({ studentId, cardToken, registered: true, filesSaved, mailSent });
}

// studentId 採番ヘルパー: idPrefix（西暦+大学コード+区分文字）で始まる既存IDの
// 連番部の最大値+1 を返す。該当なしは 1（=01）。歯抜けがあっても衝突しない。
function nextStudentSeq_(sData, sidCol, idPrefix) {
  if (sidCol < 0) return 1;
  let maxSeq = 0;
  for (let i = 1; i < sData.length; i++) {
    const s = String(sData[i][sidCol] || '');
    if (s.startsWith(idPrefix)) {
      const n = Number(s.substring(idPrefix.length));
      if (!isNaN(n) && n > maxSeq) maxSeq = n;
    }
  }
  return maxSeq + 1;
}

// 参加区分(＋ドライバー登録区分) → 属性
// ⚠ serviceClass に入るのはフォームのサービス作業クラスの選択値であり、'メカニック' という
//   文字列は送られてこない。実際の値は次のいずれか:
//     見学・応援: 'FGクラス' / '女子クラス' / 'FGクラス/女子クラス' / '実施しない'
//     補欠:       'FGクラス' / '女子クラス' / 'FGクラス/女子クラス' / '実施しない'
//   ⚠ 2026-08-17 にフォーム側の表記を統一した（app/register-pre.html）。
//     それ以前に保存された補欠の行には旧表記が残る:
//       'FG/女子クラス' / 'サービス作業は実施しない/来場予定はない'
//     下の indexOf('実施しない') 判定は新旧どちらの表記でも成立する。
//   以前は serviceClass === 'メカニック' で判定していたため常に偽になり、
//   サービス作業を行う学生も全員「応援学生」として保存されていた。
//   「実施しない」を含むかどうかで判定する（受付リスト側と同じ規則に揃える）。
function preAttr_(category, driverClass, serviceClass) {
  const svc = String(serviceClass || '');
  const doesService = !!svc && svc.indexOf('実施しない') < 0;
  switch (String(category || '')) {
    case '出場選手(FGクラスドライバー)': return String(driverClass || 'Aドライバー');
    case '出場選手(女子クラスドライバー)': return '女子クラスドライバー';
    case '補欠ドライバー':                 return '補欠ドライバー';
    case '見学・応援学生(メカニック登録含む)':
      return doesService ? 'メカニック' : '応援学生';
    default: return '応援学生';
  }
}

    'メールアドレス': p.email || '', '電話番号': p.phone || '', '郵便番号': p.postal || '',
    '都道府県': p.prefecture || '', '住所詳細': p.address || '',
    'ドライバー登録区分': p.driverClass || '', '自動車部在籍年数': p.clubYears || '',
    'シムアタック受験日': p.simDate || '', 'シムアタックタイム': p.simTime || '', '意気込み': p.comment || '',

    'サービス作業クラス': p.serviceClass || '', '予選会出場': p.competing || '',
    '規則誓約同意': p.rulesConsent === 'true' ? 'true' : 'false',
    '個人情報同意': p.privacyConsent === 'true' ? 'true' : 'false',
    'メディア同意': p.mediaConsent === 'true' ? 'true' : 'false',

    '保険証明URL': p.insuranceUrl || '', '承諾書URL': p.approvalUrl || '',

  });

}

// 当日参加者の個人ページ(氏名+QR)URLを組み立てる。
// appBase は静的サイトの app/ ディレクトリの絶対URL（フロントが location から渡す）。
// 取得できない場合は空文字（メール側でリンク無し扱い）。
function buildPassUrl_(appBase, token, eventId) {
  let base = String(appBase || '').trim();
  if (!base || !token) return '';
  if (base.charAt(base.length - 1) !== '/') base += '/';
  return base + 'mypass.html?token=' + encodeURIComponent(token) +
         '&event=' + encodeURIComponent(eventId);
}

// 管理画面: 当日参加者へ個人ページのリンクを再送信する。
function actionAdminResendWalkInMail_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event) return err_('missing_event');
  if (!p.token) return err_('missing_token');

  const student = findStudentByCardToken_(p.token, p.event);
  if (!student)        return err_('invalid_token');
  if (!student.email)  return err_('no_email');

  // ⚠ アクション名は adminResendWalkInMail のままだが、当日参加者専用ではない。
  //   事前登録者にも送れる（受付でパスを忘れた学生への送信）。
  //   改名するとフロントとGASの同時デプロイが必要になるため名前は据え置く。
  //   文面だけ区分で出し分ける: 事前登録者は「初回送信」なので専用テンプレを使う。
  const passUrl = buildPassUrl_(p.appBase, p.token, p.event);
  const sent = (String(student.regType) === '事前')
    ? sendPassMail_(student.email, student.name, p.event, passUrl)
    : sendWalkInPassMail_(student.email, student.name, p.event, passUrl);
  // ⚠ 送信できなかったら成功を返さない。受付で「送信しました」と出たのに
  //   届いていない、という状況を作らないため（2026-08-25）。
  if (!sent) return err_('mail_send_failed');
  return ok_({ sent: true, email: student.email, regType: String(student.regType || '') });
}

// 事前登録の提出ファイルをDriveに保存（非公開＝オーナー/組織のみ閲覧）。URLを返す。
// 階層: マイドライブ/FG事前登録書類/<eventId>/<書類種別>/  に保存（イベント・種別で混在防止）
function savePreRegFiles_(eventId, studentId, files, name) {
  const urls = {};
  if (!files || typeof files !== 'object') return urls;
  const eventFolder = getPreRegEventFolder_(eventId);
  const map = { pledge: '誓約書', license: '免許証', insurance: '保険証明', approval: '承諾書' };
  Object.keys(map).forEach(k => {
    const f = files[k];
    if (f && f.b64) {
      try {
        const sub   = getOrCreateChildFolder_(eventFolder, map[k]);
        const bytes = Utilities.base64Decode(f.b64);
        const fname = [studentId, name, (f.name || 'file')].filter(String).join('_');
        const blob  = Utilities.newBlob(bytes, f.mime || 'application/octet-stream', fname);
        const created = sub.createFile(blob); // 既定の権限＝非公開（個人情報のため共有しない）
        urls[k + 'Url'] = created.getUrl();
      } catch (err) { console.error('savePreRegFiles_', k, err); }
    }
  });
  return urls;
}

// 送られてきたファイルすべてについて、保存後のURLが得られているかを検証する。
// savePreRegFiles_ はファイル単位で例外を握りつぶして続行するため、
// この突き合わせをしないと「1枚だけ保存できていない」状態を検知できない。
function expectedFilesAllSaved_(files, urls) {
  if (!files || typeof files !== 'object') return true;
  const keys = ['pledge', 'license', 'insurance', 'approval'];
  for (let i = 0; i < keys.length; i++) {
    const k = keys[i];
    if (files[k] && files[k].b64 && !(urls || {})[k + 'Url']) return false;
  }
  return true;
}

// 事前登録シートの該当学生の行を探し、ファイルURL列だけを後追いで更新する。
// 行の追加自体はロック内で済ませてあるので、ここでは既存行の該当セルのみ触る。
function updatePreRegFileUrls_(ss, studentId, urls) {
  if (!urls || !Object.keys(urls).length) return;
  const sheet = ss.getSheetByName(SHEET.PRE_REG);
  if (!sheet || sheet.getLastRow() < 2) return;
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(String);
  const sidCol  = headers.indexOf('studentId');
  if (sidCol < 0) return;
  const sids = sheet.getRange(2, sidCol + 1, sheet.getLastRow() - 1, 1).getValues();
  let rowNum = -1;
  for (let i = sids.length - 1; i >= 0; i--) { // 同一IDが複数あれば最新行
    if (String(sids[i][0]) === String(studentId)) { rowNum = i + 2; break; }
  }
  if (rowNum < 0) return;
  const colMap = {
    pledgeUrl: '誓約書URL', licenseUrl: '免許証URL',
    insuranceUrl: '保険証明URL', approvalUrl: '承諾書URL',
  };
  Object.keys(colMap).forEach(k => {
    if (!urls[k]) return;
    const c = headers.indexOf(colMap[k]);
    if (c >= 0) sheet.getRange(rowNum, c + 1).setValue(urls[k]);
  });
}

// ポケベルコード上2桁: 1桁目=行(あ=1…ら=9,わ=0)、2桁目=段(あいうえお=1〜5)。
var POKEBELL_TABLE_ = {
  'あ':'11','い':'12','う':'13','え':'14','お':'15',
  'か':'21','き':'22','く':'23','け':'24','こ':'25',
  'さ':'31','し':'32','す':'33','せ':'34','そ':'35',
  'た':'41','ち':'42','つ':'43','て':'44','と':'45',
  'な':'51','に':'52','ぬ':'53','ね':'54','の':'55',
  'は':'61','ひ':'62','ふ':'63','へ':'64','ほ':'65',
  'ま':'71','み':'72','む':'73','め':'74','も':'75',
  'や':'81','ゆ':'83','よ':'85',
  'ら':'91','り':'92','る':'93','れ':'94','ろ':'95',
  'わ':'01','を':'05','ん':'00'
};
// 濁音・半濁音・小書き → 清音ベースへ正規化（同志社=ど→と と同じ扱い）
var KANA_BASE_TABLE_ = (function () {
  var m = {'ぁ':'あ','ぃ':'い','ぅ':'う','ぇ':'え','ぉ':'お','っ':'つ','ゃ':'や','ゅ':'ゆ','ょ':'よ','ゎ':'わ'};
  'がぎぐげござじずぜぞだぢづでどばびぶべぼ'.split('').forEach(function (c) {
    m[c] = String.fromCharCode(c.charCodeAt(0) - 1);
  });
  'ぱぴぷぺぽ'.split('').forEach(function (c) {
    m[c] = String.fromCharCode(c.charCodeAt(0) - 2);
  });
  return m;
})();

function pokebellPrefix_(reading) {
  if (!reading) return '';
  var c = String(reading).trim().charAt(0);
  if (c >= 'ァ' && c <= 'ヶ') c = String.fromCharCode(c.charCodeAt(0) - 0x60); // カタカナ→ひらがな
  c = KANA_BASE_TABLE_[c] || c;
  return POKEBELL_TABLE_[c] || '';
}

// 状態が「確定」でない大学（自動割当(要確認) など）に加え、既に「確定」済みでも
// 確定処理（コード確定後の学生ID一括反映）が未完了の大学を resuming:true で返す。
// ⚠ resumingの大学は codeCol の値ではなく、確定履歴の親行に記録された newCode を返す
//   （マスターへの書込が完了していない再開直後の一瞬だけ、両者が一致しないタイミングがあるため）。
function actionAdminGetPendingUniversities_(p) {
  var e = validateAdmin_(p);
  if (e) return e;
  var sheet = masterSs_().getSheetByName(SHEET.UNI_MASTER);
  if (!sheet || sheet.getLastRow() < 2) return ok_({ universities: [] });
  var data      = sheet.getDataRange().getValues();
  var headers   = data[0].map(String);
  var nameCol   = headers.indexOf('大学名');
  var codeCol   = headers.indexOf('コード');
  var statusCol = headers.indexOf('状態');
  // ⚠ 大学ごとに findActiveConfirmOperation_ を呼ぶと確定履歴シートを毎回全読みするN+1になる。
  //   履歴は1回だけ読んで「大学名→未完了操作」のMapに畳んでから突き合わせる。
  var activeByName = buildActiveConfirmOperationMap_();
  var out = [];
  data.slice(1).forEach(function (row) {
    var name   = String(row[nameCol] || '').trim();
    var status = statusCol >= 0 ? String(row[statusCol] || '').trim() : '';
    if (!name) return;
    var active = activeByName[name];
    if (status !== '確定') {
      // 未確定でも未完了の確定操作があれば「反映中」。大学マスターの更新はスイープ完了後に
      // 行うため、中断中の大学はここ（未確定）に現れる。読みの再入力は不要なので resuming 扱い。
      out.push({
        name: name,
        code: active ? active.newCode : String(row[codeCol] || '').trim(),
        status: status,
        resuming: !!active,
      });
      return;
    }
    // 確定済みなのに未完了操作が残っている＝マスター更新後に落ちたケース
    if (active) {
      out.push({ name: name, code: active.newCode, status: status, resuming: true });
    }
  });
  return ok_({ universities: out });
}

// ============================================================
// 大学確定履歴（未登録大学の「確定」処理を再開可能にするための永続化）
// 大学統合履歴と同型だが、確定フローは対応表(idMap)を持たない単純な
// コード置換（oldCode→newCode）のため、対応表JSON列の代わりに
// oldCode/newCodeの2列を持つ。親行(eventId='__operation__')が操作全体の
// 状態、子行(実イベントID)がイベント単位の進捗を表す（P1-3対策：
// 「今その時点のアクティブイベント一覧」に完了判定を依存させないことで、
// 後日イベントが増えても完了済み操作が未完了扱いに戻らないようにする）。
// ============================================================

function getConfirmHistorySheet_() {
  var sheet = masterSs_().getSheetByName(SHEET.CONFIRM_HISTORY);
  if (!sheet) {
    sheet = masterSs_().insertSheet(SHEET.CONFIRM_HISTORY);
    sheet.getRange(1, 1, 1, 7).setValues([['operationId', '大学名', 'oldCode', 'newCode', 'eventId', '状態', '更新時刻']]);
    formatHeaderRow_(sheet);
  }
  return sheet;
}

function findConfirmHistoryRow_(sheet, operationId, eventId) {
  if (sheet.getLastRow() < 2) return -1;
  var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 5).getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][0]) === operationId && String(data[i][4]) === eventId) return i + 2;
  }
  return -1;
}

function upsertConfirmHistory_(sheet, operationId, name, oldCode, newCode, eventId, status) {
  var row = findConfirmHistoryRow_(sheet, operationId, eventId);
  var values = [[operationId, name, oldCode, newCode, eventId, status, new Date()]];
  if (row < 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, 7).setValues(values);
  } else {
    sheet.getRange(row, 1, 1, 7).setValues(values);
  }
}

// 大学名から、未完了(親行が処理中／要確認)の確定操作を検索する。
// ⚠ 大学マスターの現在の状態(確定/未確定)を一切見ない。履歴保存後・マスター更新前に
//   失敗した場合でも、この検索結果だけを根拠に再開できるようにするため（P0-1対策）。
function findActiveConfirmOperation_(historySheet, name) {
  if (historySheet.getLastRow() < 2) return null;
  var data = historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 6).getValues();
  for (var i = data.length - 1; i >= 0; i--) {
    if (String(data[i][1]) === name && String(data[i][4]) === '__operation__' &&
        (String(data[i][5]) === '処理中' || String(data[i][5]) === '要確認')) {
      return { operationId: String(data[i][0]), oldCode: String(data[i][2]), newCode: String(data[i][3]) };
    }
  }
  return null;
}

// 未完了(親行が処理中／要確認)の確定操作を、大学名をキーにしたオブジェクトへ畳んで返す。
// 一覧表示のように多数の大学を一度に判定する用途では、大学ごとに
// findActiveConfirmOperation_ を呼ぶと履歴シートを毎回全読みするN+1になるため、
// 必ずこちらを使って1回の読み込みで済ませる。
// 同じ大学に複数の未完了操作があった場合は、後の行（＝新しい操作）を優先する。
function buildActiveConfirmOperationMap_() {
  var map = {};
  var historySheet = masterSs_().getSheetByName(SHEET.CONFIRM_HISTORY); // 未作成なら未完了操作は無い(新規作成はしない)
  if (!historySheet || historySheet.getLastRow() < 2) return map;
  var data = historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 6).getValues();
  data.forEach(function (row) {
    if (String(row[4]) !== '__operation__') return;
    var status = String(row[5]);
    if (status !== '処理中' && status !== '要確認') return;
    map[String(row[1])] = { operationId: String(row[0]), oldCode: String(row[2]), newCode: String(row[3]) };
  });
  return map;
}

// 他大学分も含め、未完了の確定操作が予約済みのnewCodeを全て集める。
// ⚠ 大学マスターへの書込前に履歴だけ保存された状態でクラッシュすると、ロックは
//   スクリプト終了時に自動解放されるが予約は履歴に残る。次の確定処理がusedCodesを
//   大学マスターの現在値だけから計算すると、この孤立予約と同じコードを
//   再度採番しうる（P0-4対策）。
function collectReservedCodes_(historySheet) {
  var reserved = new Set();
  if (historySheet.getLastRow() < 2) return reserved;
  var data = historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 6).getValues();
  data.forEach(function (row) {
    if (String(row[4]) !== '__operation__') return;
    var status = String(row[5]);
    if (status === '処理中' || status === '要確認') reserved.add(String(row[3]).padStart(4, '0'));
  });
  return reserved;
}

// 大学マスターから大学名で1行を探す（確定フロー用。統合の findMergeUniversityRows_ と
// 違い対象は1件のみで、統合先を持たない）。
function findUniversityRowByName_(name) {
  var sheet = masterSs_().getSheetByName(SHEET.UNI_MASTER);
  if (!sheet || sheet.getLastRow() < 2) return { found: false };
  var data      = sheet.getDataRange().getValues();
  var headers   = data[0].map(String);
  var nameCol   = headers.indexOf('大学名');
  var codeCol   = headers.indexOf('コード');
  var statusCol = headers.indexOf('状態');
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][nameCol] || '').trim() === name) {
      var raw = String(data[i][codeCol] || '').trim().replace(/[０-９]/g, function(c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); });
      return {
        found: true, sheet: sheet, row: i + 1,
        code: raw.padStart(4, '0'),
        status: statusCol >= 0 ? String(data[i][statusCol] || '').trim() : '',
        codeCol: codeCol, statusCol: statusCol,
      };
    }
  }
  return { found: false };
}

// シート全体ではなく studentId 列のデータ範囲だけを対象に一括読み書きする。
// ⚠ getDataRange()で全列を対象にすると、事務局が手動追加した数式列があった場合に
//   静的な値へ上書きしてしまうリスクがある（大学統合機能のapplyUniMergeToSheet_に
//   同種のリスクが残っている。docs/NOTES.md参照）。新規実装ではこれを避ける。
function rewriteStudentUniCodeInSheet_(ss, sheetName, oldCode, newCode) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet || sheet.getLastRow() < 2) return 0;
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(String);
  var sidCol  = headers.indexOf('studentId');
  if (sidCol < 0) return 0;
  var range = sheet.getRange(2, sidCol + 1, sheet.getLastRow() - 1, 1);
  var vals  = range.getValues();
  var updated = 0;
  for (var i = 0; i < vals.length; i++) {
    var s = String(vals[i][0] || '');
    if (s.length >= 5 && s.substring(1, 5) === oldCode) {
      vals[i][0] = "'" + s.charAt(0) + newCode + s.substring(5); // 先頭0保持のため文字列化
      updated++;
    }
  }
  if (updated > 0) range.setValues(vals);
  return updated;
}

// CONFIGのwomenPairings内、a/b/cのstudentIdコード部を置換する（確定フロー用。
// 統合機能のapplyUniMergeToConfig_と違い対応表(idMap)ではなくoldCode/newCodeの
// 単純な部分置換）。
function rewriteWomenPairingsCode_(ss, oldCode, newCode) {
  var sheet = ss.getSheetByName(SHEET.CONFIG);
  if (!sheet || sheet.getLastRow() < 1) return;
  var data = sheet.getDataRange().getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][0]) !== 'womenPairings') continue;
    var raw = String(data[i][1] || '');
    if (!raw) return;
    try {
      var arr = JSON.parse(raw);
      if (!Array.isArray(arr)) return;
      var changed = false;
      var fix = function (sid) {
        if (sid && sid.length >= 5 && sid.substring(1, 5) === oldCode) {
          changed = true;
          return sid.charAt(0) + newCode + sid.substring(5);
        }
        return sid;
      };
      arr.forEach(function (pair) {
        if (!pair) return;
        pair.a = fix(String(pair.a || ''));
        pair.b = fix(String(pair.b || ''));
        pair.c = fix(String(pair.c || ''));
      });
      if (changed) sheet.getRange(i + 1, 2).setValue(JSON.stringify(arr));
    } catch (err) { /* 壊れたJSONは触らない */ }
    return;
  }
}

// 読みの頭文字からポケベルコードを生成し、既存コードを見て連番採番して確定する。
// 再開可能な設計:
//   1. 大学名から未完了の確定操作を状態に関係なく必ず先に検索する。見つかればその
//      oldCode/newCodeを再利用し、新規採番は絶対に行わない（履歴保存後・マスター
//      更新前の失敗からも安全に再開できる＝P0-1対策）。
//   2. 見つからない場合のみ新規採番する。他大学の未完了操作が予約済みのnewCodeも
//      使用済み扱いにする（P0-4対策）。
//   3. 書き込み順序は「履歴の親行 → 全イベントのスイープ → 最後に大学マスター」。
//      大学マスターを先に新コードへ更新してしまうと、スイープが中断した隙に入った
//      新規登録が新コードの連番を取り、再開時に旧コードの学生が同じIDへ変換されて
//      衝突する。マスターを最後にすることで、中断中の登録も旧コードで採番され、
//      再開時のスイープで一緒に拾える。
//   4. ロックはスイープ完了までスクリプト全体で保持する。学生登録
//      (actionRegisterPreStudent_)も同じLockService.getScriptLock()を使うため、
//      スイープ中の新規登録は自然にブロックされ、途中登録によるID衝突を防げる
//      （大学統合機能で既に受容済みのトレードオフを踏襲。登録側は変更しない＝P0-2対策）。
//   5. 非アクティブなイベントは対象外（統合機能と仕様を一致）。
//   6. 親行(eventId='__operation__')は全子行が完了/対象外で揃って初めて「完了」にする。
//      対象イベント0件でも完了扱いにできる（P1-3対策）。
function actionAdminConfirmUniversity_(p) {
  var e = validateAdmin_(p);
  if (e) return e;
  var name = String(p.name || '').trim();
  if (!name) return err_('missing_params');

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return err_('locked');
  try {
    var historySheet = getConfirmHistorySheet_();
    var active = findActiveConfirmOperation_(historySheet, name); // 状態に関係なく必ず先に検索(P0-1対策)

    var uni = findUniversityRowByName_(name);
    if (!uni.found) return err_('university_not_found');

    var operationId, oldCode, newCode;

    if (active) {
      operationId = active.operationId;
      oldCode     = active.oldCode;
      newCode     = active.newCode;
    } else {
      if (uni.status === '確定') return err_('already_confirmed_no_history'); // 履歴の無い確定済み(旧データ)は安全側で停止
      var prefix = pokebellPrefix_(p.reading);
      if (!prefix) return err_('invalid_reading');

      oldCode = uni.code;
      var reserved = collectReservedCodes_(historySheet); // P0-4対策
      var maxSeq = 0;
      var uniData = uni.sheet.getDataRange().getValues();
      uniData.slice(1).forEach(function (row) {
        var raw = String(row[uni.codeCol] || '').trim().replace(/[０-９]/g, function(c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); });
        var code = raw.padStart(4, '0');
        reserved.add(code);
        if (code.substring(0, 2) === prefix) {
          var seq = Number(code.substring(2));
          if (!isNaN(seq) && seq > maxSeq) maxSeq = seq;
        }
      });
      var nextSeq = maxSeq + 1;
      while (reserved.has(prefix + String(nextSeq).padStart(2, '0'))) { nextSeq++; }
      newCode = prefix + String(nextSeq).padStart(2, '0');

      operationId = Utilities.getUuid();
      upsertConfirmHistory_(historySheet, operationId, name, oldCode, newCode, '__operation__', '処理中'); // 先に履歴保存(P0-1対策)
    }

    var rewritten = 0;
    var needsReview = false;
    if (oldCode !== newCode) {
      var evSheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
      var evData  = (evSheet && evSheet.getLastRow() >= 2) ? evSheet.getDataRange().getValues() : [];
      var evIdCol = evData.length ? evData[0].map(String).indexOf('eventId') : -1;

      evData.slice(1).forEach(function (evRow) {
        var eventId = String(evRow[evIdCol] || '').trim();
        if (!eventId) return;

        if (isEventInactive_(eventId)) {
          upsertConfirmHistory_(historySheet, operationId, name, oldCode, newCode, eventId, '対象外');
          return;
        }
        // ⚠ 既に「完了」のイベントでもスキップしない。中断中(要確認のまま放置された間)は
        //   大学マスターがまだ旧コードなので、新規登録は旧コードで採番される。その学生が
        //   完了済みイベントに入る可能性があるため、毎回スイープし直す。
        //   rewriteStudentUniCodeInSheet_ は旧コードに一致する行しか触らないので、
        //   再実行しても二重変換にはならない（冪等）。
        try {
          var ss = eventSs_(eventId);
          var n = 0;
          [SHEET.STUDENTS, SHEET.STAMP_LOG, SHEET.STAMP_PARTICIPANTS, SHEET.PRIZE_LOG, SHEET.PRE_REG, SHEET.VIEW_LOG].forEach(function (sheetName) {
            n += rewriteStudentUniCodeInSheet_(ss, sheetName, oldCode, newCode);
          });
          rewriteWomenPairingsCode_(ss, oldCode, newCode);
          CacheService.getScriptCache().remove('cfg_' + eventId); // CONFIGを書き換えたらキャッシュを捨てる
          upsertConfirmHistory_(historySheet, operationId, name, oldCode, newCode, eventId, '完了');
          rewritten += n;
        } catch (err) {
          upsertConfirmHistory_(historySheet, operationId, name, oldCode, newCode, eventId, '要確認');
          needsReview = true;
        }
      });
    }

    // 親行の完了判定: この操作の全子行が 完了/対象外 で揃っていれば完了、要確認が1つでもあれば要確認
    var allChildren = historySheet.getLastRow() >= 2
      ? historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 6).getValues()
          .filter(function (r) { return String(r[0]) === operationId && String(r[4]) !== '__operation__'; })
      : [];
    var anyReview = needsReview || allChildren.some(function (r) { return String(r[5]) === '要確認'; });
    var allDone   = allChildren.length === 0 || allChildren.every(function (r) { return String(r[5]) === '完了' || String(r[5]) === '対象外'; });

    // ⚠ 全イベントのスイープが終わって初めて大学マスターを新コードへ更新する（P0対策）。
    //   先にマスターを更新すると、中断中の新規登録が新コードの連番を取ってしまい、
    //   再開時に旧コードの学生が同じIDへ変換されて衝突する。
    //   マスターが旧コードのままなら、中断中の登録も旧コードで採番され、再開時に一緒に拾える。
    //   これは大学統合が「全イベント完了後に統合元を削除する」のと同じ考え方。
    if (!anyReview && allDone) {
      var fresh = findUniversityRowByName_(name); // 行番号を取り直してから書く
      if (fresh.found) {
        fresh.sheet.getRange(fresh.row, fresh.codeCol + 1).setValue("'" + newCode);
        if (fresh.statusCol >= 0) fresh.sheet.getRange(fresh.row, fresh.statusCol + 1).setValue('確定');
      }
    }
    upsertConfirmHistory_(historySheet, operationId, name, oldCode, newCode, '__operation__', anyReview ? '要確認' : (allDone ? '完了' : '処理中'));

    return ok_({ code: newCode, rewritten: rewritten, needsReview: anyReview });
  } finally {
    lock.releaseLock();
  }
}

// 大学マスターを読み、統合元(name)・統合先(target)の行情報をまとめて返す。
// ロック取得後に必ず呼び直すこと（直前に読んだ状態を信頼しない＝P0-2対策）。
function findMergeUniversityRows_(name, target) {
  var sheet = masterSs_().getSheetByName(SHEET.UNI_MASTER);
  if (!sheet || sheet.getLastRow() < 2) return { found: false };
  var data      = sheet.getDataRange().getValues();
  var headers   = data[0].map(String);
  var nameCol   = headers.indexOf('大学名');
  var codeCol   = headers.indexOf('コード');
  var statusCol = headers.indexOf('状態');

  var srcRow = -1, srcStatus = '', dstCode = '', dstStatus = '';
  data.slice(1).forEach(function (row, i) {
    var rowName = String(row[nameCol] || '').trim();
    if (rowName === name) {
      srcRow    = i + 2;
      srcStatus = statusCol >= 0 ? String(row[statusCol] || '').trim() : '';
    }
    if (rowName === target) {
      var raw = String(row[codeCol] || '').trim().replace(/[０-９]/g, function(c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); });
      dstCode   = raw.padStart(4, '0');
      dstStatus = statusCol >= 0 ? String(row[statusCol] || '').trim() : '';
    }
  });
  if (srcRow < 0 || !dstCode) return { found: false };
  return { found: true, sheet: sheet, srcRow: srcRow, srcStatus: srcStatus, dstCode: dstCode, dstStatus: dstStatus };
}

function getMergeHistorySheet_() {
  var sheet = masterSs_().getSheetByName(SHEET.MERGE_HISTORY);
  if (!sheet) {
    sheet = masterSs_().insertSheet(SHEET.MERGE_HISTORY);
    sheet.getRange(1, 1, 1, 7).setValues([['operationId', '統合元', '統合先', 'eventId', '状態', '対応表', '更新時刻']]);
    formatHeaderRow_(sheet);
  }
  return sheet;
}

function findMergeHistoryRow_(sheet, operationId, eventId) {
  if (sheet.getLastRow() < 2) return -1;
  var data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 4).getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][0]) === operationId && String(data[i][3]) === eventId) return i + 2;
  }
  return -1;
}

function upsertMergeHistory_(sheet, operationId, name, target, eventId, status, mapJson) {
  var row = findMergeHistoryRow_(sheet, operationId, eventId);
  var values = [[operationId, name, target, eventId, status, mapJson || '', new Date()]];
  if (row < 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, 7).setValues(values);
  } else {
    sheet.getRange(row, 1, 1, 7).setValues(values);
  }
}

// 同じ統合元/統合先の組み合わせで、まだ完了していない operationId が無いか探す。
// 見つかればそれを使い続けることで、UI側にIDを持たせなくても「統合」を再度押すだけで
// 自動的に前回の続きから再開できる（見つからなければ新しい operationId を発行する）。
// 統合元(name)に対して未完了の操作が無いか探す（統合先は問わない）。
// 見つかった場合、その統合先が今回指定された target と違えば、呼び出し元は処理を拒否する
// （同じ統合元に対して途中から別の統合先を選び直せてしまい、イベントごとに
// 　統合先が分裂するのを防ぐ＝P1-4対策）。
function findActiveMergeOperation_(historySheet, name) {
  if (historySheet.getLastRow() < 2) return null;
  var data = historySheet.getRange(2, 1, historySheet.getLastRow() - 1, 5).getValues();
  for (var i = data.length - 1; i >= 0; i--) {
    if (String(data[i][1]) === name &&
        (String(data[i][4]) === '処理中' || String(data[i][4]) === '要確認')) {
      return { operationId: String(data[i][0]), target: String(data[i][2]) };
    }
  }
  return null;
}

// 指定イベントの学生マスターを見て、統合元大学(oldName)に属する未対応の学生に、
// 統合先大学コード(newCode)配下の新しいstudentIdを割り当てる。
// 西暦・クラス文字は維持し、連番だけ振り直す（統合先に既存の学生がいても衝突しない）。
//
// existingMap に既にある学生は絶対に採番し直さない（キーをそのまま引き継ぐ）。
// これにより、完了扱いのイベントでも安全に再チェックできる＝統合の処理中に新規登録された
// 学生を後から拾える（P0-1対策）。連番の起点は、実データ上の使用状況に加えて
// existingMap内の(まだ書き込まれていないかもしれない)予約済みIDも考慮するため、
// 保存直後・書き込み前のタイミングでも衝突しない。
// 戻り値: { 旧studentId: 新studentId, ... }（existingMapを含む全件）
function buildOrExtendUniMergeIdMap_(ss, oldName, newCode, existingMap) {
  var map = {};
  for (var k in existingMap) map[k] = existingMap[k];

  var sheet = ss.getSheetByName(SHEET.STUDENTS);
  if (!sheet || sheet.getLastRow() < 2) return map;
  var data    = sheet.getDataRange().getValues();
  var headers = data[0].map(String);
  var sidCol  = headers.indexOf('studentId');
  var nameCol = headers.indexOf('大学名');
  if (sidCol < 0 || nameCol < 0) return map;

  var maxSeq = {};
  for (var i = 1; i < data.length; i++) {
    var sid = String(data[i][sidCol] || '');
    if (sid.length < 8 || sid.substring(1, 5) !== newCode) continue;
    var prefix = sid.substring(0, 6);
    var seq    = Number(sid.substring(6)) || 0;
    if (!maxSeq[prefix] || seq > maxSeq[prefix]) maxSeq[prefix] = seq;
  }
  // 対応表に既にある(まだ実データへ未反映かもしれない)新IDも予約済みとして扱う
  for (var k2 in existingMap) {
    var reservedSid = String(existingMap[k2]);
    if (reservedSid.length < 8) continue;
    var rPrefix = reservedSid.substring(0, 6);
    var rSeq    = Number(reservedSid.substring(6)) || 0;
    if (!maxSeq[rPrefix] || rSeq > maxSeq[rPrefix]) maxSeq[rPrefix] = rSeq;
  }

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][nameCol] || '').trim() !== oldName) continue;
    var oldSid = String(data[i][sidCol] || '');
    if (oldSid.length < 8 || map[oldSid]) continue; // 想定外の形式・既に対応表にある学生はスキップ
    var prefix = oldSid.charAt(0) + newCode + oldSid.charAt(5); // 西暦+統合先コード+クラス文字は維持
    var nextSeq = (maxSeq[prefix] || 0) + 1;
    maxSeq[prefix] = nextSeq;
    map[oldSid] = prefix + String(nextSeq).padStart(2, '0');
  }
  return map;
}

// 指定シートに対し、studentId をマップに従って書き換え、大学名テキストも統合先へ書き換える。
// シート単位で一括読み込み・一括書き込みする（セル単位のAPI呼び出しを避けてロック保持時間を縮める）。
// マップに無い(既に書き換え済みの)行は自然にスキップされるため、この関数自体が
// 再実行しても安全＝冪等になっている。
// ⚠ 以前は getDataRange() でシート全列を読み、setValues() で全列を書き戻していた。
//   getValues() は数式ではなく計算結果を返すため、事務局が手作業で追加した数式列が
//   あると、統合の実行で静的な値に置き換わって失われる恐れがあった。
//   実際に書き換える studentId 列・大学名 列だけを個別に読み書きする。
function applyUniMergeToSheet_(ss, sheetName, idMap, oldName, newName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet || sheet.getLastRow() < 2) return 0;
  var lastRow = sheet.getLastRow();
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(String);
  var sidCol  = headers.indexOf('studentId');
  var nameCol = headers.indexOf('大学名');
  if (sidCol < 0 && nameCol < 0) return 0;

  var touched = {}; // 影響を受けた行（0始まり）。戻り値の件数を従来と揃えるため

  if (sidCol >= 0) {
    var sidRange = sheet.getRange(2, sidCol + 1, lastRow - 1, 1);
    var sids = sidRange.getValues();
    var changedSid = 0;
    for (var i = 0; i < sids.length; i++) {
      var newSid = idMap[String(sids[i][0] || '')];
      if (newSid) { sids[i][0] = "'" + newSid; touched[i] = true; changedSid++; }
    }
    if (changedSid > 0) sidRange.setValues(sids);
  }

  if (nameCol >= 0) {
    var nameRange = sheet.getRange(2, nameCol + 1, lastRow - 1, 1);
    var names = nameRange.getValues();
    var changedName = 0;
    for (var j = 0; j < names.length; j++) {
      if (String(names[j][0] || '').trim() === oldName) {
        names[j][0] = newName; touched[j] = true; changedName++;
      }
    }
    if (changedName > 0) nameRange.setValues(names);
  }

  return Object.keys(touched).length;
}

// 出場校エントリーの学校名を統合先へ書き換える。
// 統合元・統合先の両方が既に提出済みの場合、どちらが正しいかは自動判定しない
// （日付だけで機械的に決めると、古い方に正しい情報が残っている可能性を潰しかねない）。
// 単純にテキストを置換し、重複が生じたことだけを呼び出し元へ伝える。
// 実際にどちらを残すかは、内容を見た上でスタッフが手動で判断・削除する。
function reconcileSchoolEntry_(ss, oldName, newName) {
  var sheet = ss.getSheetByName(SHEET.SCHOOL_ENTRIES);
  if (!sheet || sheet.getLastRow() < 2) return false;
  var data    = sheet.getDataRange().getValues();
  var headers = data[0].map(String);
  var nameCol = headers.indexOf('学校名');
  if (nameCol < 0) return false;

  var oldRow = -1, alreadyHasTarget = false;
  for (var i = 1; i < data.length; i++) {
    var n = String(data[i][nameCol] || '').trim();
    if (n === oldName) oldRow = i + 1;
    if (n === newName) alreadyHasTarget = true;
  }
  if (oldRow < 0) return false; // 統合元の提出が無ければ何もしない

  sheet.getRange(oldRow, nameCol + 1).setValue(newName);
  return alreadyHasTarget; // true = 重複が発生した(要手動確認)
}

// 対象イベントのCONFIGを更新する。womenPairings(studentId)は新IDへ、
// schoolRunningOrder(大学名)は統合先名へ、それぞれ位置を保ったまま追従させる。
function applyUniMergeToConfig_(ss, idMap, oldName, newName) {
  var sheet = ss.getSheetByName(SHEET.CONFIG);
  if (!sheet || sheet.getLastRow() < 1) return;
  var data = sheet.getDataRange().getValues();
  for (var i = 0; i < data.length; i++) {
    var key = String(data[i][0]);
    if (key === 'womenPairings' && Object.keys(idMap).length) {
      var raw = String(data[i][1] || '');
      if (!raw) continue;
      try {
        var arr = JSON.parse(raw);
        if (!Array.isArray(arr)) continue;
        var changed = false;
        arr.forEach(function (pair) {
          if (!pair) return;
          if (idMap[pair.a]) { pair.a = idMap[pair.a]; changed = true; }
          if (idMap[pair.b]) { pair.b = idMap[pair.b]; changed = true; }
          if (idMap[pair.c]) { pair.c = idMap[pair.c]; changed = true; }
        });
        if (changed) sheet.getRange(i + 1, 2).setValue(JSON.stringify(arr));
      } catch (err) { /* 壊れたJSONは触らない */ }
    }
    if (key === 'schoolRunningOrder') {
      var raw2 = String(data[i][1] || '');
      if (!raw2) continue;
      try {
        var arr2 = JSON.parse(raw2);
        if (!Array.isArray(arr2)) continue;
        var idx = arr2.indexOf(oldName);
        if (idx >= 0) {
          arr2[idx] = newName;
          // 置換の結果、統合先が既に別の位置にもあれば重複除去する(先勝ち)
          // ＝重複したまま保存すると大学数Nが水増しされ全クラスの走行順がずれる(P1-5対策)
          var seen = {};
          var deduped = arr2.filter(function (nm) {
            if (seen[nm]) return false;
            seen[nm] = true;
            return true;
          });
          sheet.getRange(i + 1, 2).setValue(JSON.stringify(deduped));
        }
      } catch (err) { /* 壊れたJSONは触らない */ }
    }
  }
}

// 統合の影響件数だけを確認する（書き込み・ロックともに一切行わない）。
// ⚠ 旧GASにはこのアクション自体が存在しないため、フロントが新しいプレビューの
//   仕組みを前提にしていても、旧GASでは unknown_action で安全に停止する
//   （dryRunフラグを旧GASの実処理アクションへ渡して無視される事故を防ぐため、
//   意図的に別アクション名にしている＝P0-1対策）。
function actionAdminPreviewUniversityMerge_(p) {
  var e = validateAdmin_(p);
  if (e) return e;
  var name   = String(p.name || '').trim();
  var target = String(p.target || '').trim();
  if (!name || !target) return err_('missing_params');
  if (name === target) return err_('same_university');

  var uni = findMergeUniversityRows_(name, target);
  if (!uni.found) return err_('university_not_found');
  if (uni.dstStatus !== '確定') return err_('target_not_confirmed');

  var evSheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
  if (!evSheet || evSheet.getLastRow() < 2) return ok_({ students: 0, events: 0 });
  var evData  = evSheet.getDataRange().getValues();
  var evIdCol = evData[0].map(String).indexOf('eventId');

  var totalStudents = 0, affectedEvents = 0;
  evData.slice(1).forEach(function (evRow) {
    var eventId = String(evRow[evIdCol] || '').trim();
    if (!eventId || isEventInactive_(eventId)) return; // 非アクティブは対象外(件数にも数えない・仕様)
    var ss;
    try { ss = eventSs_(eventId); } catch (err) { return; }
    var idMap = buildOrExtendUniMergeIdMap_(ss, name, uni.dstCode, {});
    var n = Object.keys(idMap).length;
    if (n > 0) { totalStudents += n; affectedEvents++; }
  });
  return ok_({ students: totalStudents, events: affectedEvents });
}

// 表記ゆれで生まれた大学を、既存の確定済み大学へ統合する。
// ⚠ 統合先の判定は一切自動化しない。呼び出し元(管理画面)でスタッフが手動選択した
//   name(統合元) と target(統合先) をそのまま信頼して実行する（サーバー側では
//   統合元が未確定・統合先が確定済みであることを、ロック取得後に再検証する）。
//
// 再開可能な設計:
//   1. イベント単位の「旧ID→新ID対応表」を大学統合履歴シートへ先に永続化してから、
//      実際のデータシートを書き換える。既存の対応表エントリは絶対に採番し直さない。
//   2. 「完了」を恒久的なスキップとはせず、大学マスターに統合元が残っている限り
//      毎回残存学生の有無を再チェックする。途中で新規登録された学生も、
//      次回実行時に対応表へ追加され取りこぼされない（P0-1対策）。
//   3. 接続例外時・保存済み対応表が破損していた時は、既存の対応表を消さずに
//      状態だけ「要確認」にする（P0-2対策）。
//   4. 全イベントが完了（または非アクティブとして対象外）して初めて、大学マスターから
//      統合元を削除する。接続例外が1件でも残っていれば、統合元は消さずに残す。
//   5. 非アクティブなイベントは意図的に対象外とする（仕様。CLAUDE.md参照）。
//      接続できないだけの「要確認」とは区別して記録する。
//   6. 統合元に対して既に別の統合先への未完了操作がある場合は拒否する（P1-4対策）。
function actionAdminMergeUniversity_(p) {
  var e = validateAdmin_(p);
  if (e) return e;
  var name   = String(p.name || '').trim();
  var target = String(p.target || '').trim();
  if (!name || !target) return err_('missing_params');
  if (name === target) return err_('same_university');

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return err_('locked');
  try {
    // ロック取得後に必ず再読込する（判定とロックの間に他の操作が割り込む余地を無くす＝P0-2対策）
    var uni = findMergeUniversityRows_(name, target);
    if (!uni.found) return err_('university_not_found');
    if (uni.srcStatus === '確定') return err_('source_already_confirmed'); // P2-8対策
    if (uni.dstStatus !== '確定') return err_('target_not_confirmed');

    var historySheet = getMergeHistorySheet_();
    var active = findActiveMergeOperation_(historySheet, name);
    if (active && active.target !== target) return err_('merge_in_progress_different_target'); // P1-4対策
    var operationId = active ? active.operationId : Utilities.getUuid();

    var evSheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
    if (!evSheet || evSheet.getLastRow() < 2) return err_('no_events');
    var evData  = evSheet.getDataRange().getValues();
    var evIdCol = evData[0].map(String).indexOf('eventId');

    var totalStudents  = 0;
    var affectedEvents = 0;
    var needsReview    = false; // 接続例外・対応表破損が1件でもあれば true（統合元を消さず残す）
    var schoolEntryDup = false;

    evData.slice(1).forEach(function (evRow) {
      var eventId = String(evRow[evIdCol] || '').trim();
      if (!eventId) return;

      if (isEventInactive_(eventId)) {
        upsertMergeHistory_(historySheet, operationId, name, target, eventId, '対象外', '');
        return;
      }

      var existingRow     = findMergeHistoryRow_(historySheet, operationId, eventId);
      var existingStatus  = existingRow > 0 ? String(historySheet.getRange(existingRow, 5).getValue()) : '';
      var existingMapJson = existingRow > 0 ? String(historySheet.getRange(existingRow, 6).getValue() || '') : '';

      var existingMap = {};
      if (existingMapJson) {
        try {
          var parsed = JSON.parse(existingMapJson);
          if (!parsed || typeof parsed !== 'object') throw new Error('invalid');
          existingMap = parsed;
        } catch (errParse) {
          // 保存済み対応表が壊れている場合は完了扱いにせず要確認で止める（消さない＝P0-2対策）
          upsertMergeHistory_(historySheet, operationId, name, target, eventId, '要確認', existingMapJson);
          needsReview = true;
          return;
        }
      }

      var ss;
      try { ss = eventSs_(eventId); } catch (err) {
        // 既存の対応表があれば保持したまま状態だけ要確認にする（空文字で消さない＝P0-2対策）
        upsertMergeHistory_(historySheet, operationId, name, target, eventId, '要確認', existingMapJson);
        needsReview = true;
        return;
      }

      // 完了済みでも、統合元がまだ大学マスターに残っている＝この関数が呼ばれている間は、
      // その間に新規登録された学生が無いか毎回確認する。既存の対応表のキーは
      // buildOrExtendUniMergeIdMap_ 内で必ず維持されるため、採番がずれることは無い。
      var idMap = buildOrExtendUniMergeIdMap_(ss, name, uni.dstCode, existingMap);
      var newlyMapped = Object.keys(idMap).length - Object.keys(existingMap).length;
      if (existingStatus === '完了' && newlyMapped === 0) return; // 本当に完了・追加なし

      upsertMergeHistory_(historySheet, operationId, name, target, eventId, '処理中', JSON.stringify(idMap));

      var n = Object.keys(idMap).length;
      if (n > 0) affectedEvents++;
      totalStudents += n;

      // 学生マスターは最後に書き換える。対応表自体は既に大学統合履歴へ保存済みなので、
      // 途中で失敗してもそちらを使って安全に再開できる。
      applyUniMergeToSheet_(ss, SHEET.PRE_REG,            idMap, name, target);
      applyUniMergeToSheet_(ss, SHEET.VIEW_LOG,           idMap, name, target);
      applyUniMergeToSheet_(ss, SHEET.STAMP_LOG,          idMap, name, target);
      applyUniMergeToSheet_(ss, SHEET.STAMP_PARTICIPANTS, idMap, name, target);
      applyUniMergeToSheet_(ss, SHEET.PRIZE_LOG,          idMap, name, target);
      if (reconcileSchoolEntry_(ss, name, target)) schoolEntryDup = true;
      applyUniMergeToConfig_(ss, idMap, name, target);
      CacheService.getScriptCache().remove('cfg_' + eventId); // CONFIGを書き換えたらキャッシュを捨てる
      applyUniMergeToSheet_(ss, SHEET.STUDENTS,           idMap, name, target);

      upsertMergeHistory_(historySheet, operationId, name, target, eventId, '完了', JSON.stringify(idMap));
    });

    if (needsReview) {
      return ok_({
        students: totalStudents, events: affectedEvents,
        schoolEntryDup: schoolEntryDup, needsReview: true,
      });
    }

    // 全イベントが完了（または非アクティブとして対象外）して初めて、統合元を削除する。
    // ⚠ 意図的に「統合した」という記録を残さない。記録を残して次回以降に自動解決させると、
    //   同じ表記が再入力されたときに人の目に触れないまま同一大学として扱われてしまう。
    //   同じ表記が再入力されたら、また承認待ちに上がって毎回ゼロから人が判断する。
    var fresh = findMergeUniversityRows_(name, target); // 削除直前に行番号を再取得（P0-2対策）
    if (fresh.found) fresh.sheet.deleteRow(fresh.srcRow);

    return ok_({
      students: totalStudents, events: affectedEvents,
      schoolEntryDup: schoolEntryDup, needsReview: false,
    });
  } finally {
    lock.releaseLock();
  }
}

// 学生情報を全シート横断で一括修正（氏名/ふりがな/大学名/属性/メールアドレス）
// ⚠ 対象は p.event のイベント1件のみ。
//   studentId は「西暦1桁+大学コード4桁+区分1桁+連番2桁」で、連番はイベント別シート内で
//   採番されるため、同じ年度の別イベントに同一IDの別人が実在する。
//   以前は全イベントを走査していたため、別イベントの別人の氏名・大学・メールまで
//   書き換わっていた（P0）。
//
// 大学名・属性を変更した場合は studentId を新しい大学コード/区分で採番し直し、
// studentId を持つ全シートと CONFIG.womenPairings を追従させる。
//   - cardToken は据え置く（名札QRは cardToken を埋め込んでいるため、
//     既に印刷・配布済みのQRがそのまま使え、表示されるIDだけが新しくなる）
//   - 採番と書き換えは学生登録と同じスクリプトロック内で行い、同時登録との衝突を防ぐ
function actionAdminUpdateStudent_(p) {
  var e = validateAdmin_(p);
  if (e) return e;

  var sid = String(p.studentId || '').trim();
  if (!sid) return err_('missing_params');
  if (!p.event) return err_('missing_event');
  var eventId = String(p.event).trim();

  // 渡されたフィールドのみ更新
  var fields = {};
  if (p.name     !== undefined) fields.name     = String(p.name).trim();
  if (p.furigana !== undefined) fields.furigana  = String(p.furigana).trim();
  if (p.school   !== undefined) fields.school    = String(p.school).trim();
  if (p.category !== undefined) fields.category  = String(p.category).trim();
  if (p.email    !== undefined) fields.email     = String(p.email).trim();
  if (!Object.keys(fields).length) return err_('missing_params');

  // 属性は自由入力のため、未知の値だと区分コードが既定値'V'に落ちて静かに誤分類される。
  // 採番し直す前に既知の属性かどうかを検証して弾く。
  if (fields.category && PRE_CLASS_CODES[fields.category] === undefined) {
    return err_('unknown_category');
  }

  // 各シートの列名マッピング
  var masterMap = { name:'氏名', furigana:'ふりがな', school:'大学名', category:'属性', email:'メールアドレス' };
  var preMap    = { name:'氏名', furigana:'ふりがな', school:'大学名', email:'メールアドレス' }; // category は事前登録には書かない

  var updated = { students: 0, preReg: 0, stampLog: 0, prizeLog: 0 };
  var eventIds = [eventId];

  function updateSheet_(sheet, hdrs, data, sidCol, colMap) {
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][sidCol]) !== sid) continue;
      Object.keys(colMap).forEach(function(key) {
        if (fields[key] === undefined) return;
        var c = hdrs.indexOf(colMap[key]);
        if (c >= 0) sheet.getRange(i + 1, c + 1).setValue(fields[key]);
      });
      return true;
    }
    return false;
  }

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) return err_('locked');
  try {

  eventIds.forEach(function(eventId) {
    var ss;
    try { ss = eventSs_(eventId); } catch (e2) { return; }

    // 学生マスター
    var stuSheet = ss.getSheetByName(SHEET.STUDENTS);
    if (stuSheet && stuSheet.getLastRow() > 1) {
      var d = stuSheet.getDataRange().getValues();
      var h = d[0].map(String);
      if (updateSheet_(stuSheet, h, d, h.indexOf('studentId'), masterMap)) updated.students++;
    }

    // 事前登録
    var preSheet = ss.getSheetByName(SHEET.PRE_REG);
    if (preSheet && preSheet.getLastRow() > 1) {
      var d = preSheet.getDataRange().getValues();
      var h = d[0].map(String);
      if (updateSheet_(preSheet, h, d, h.indexOf('studentId'), preMap)) updated.preReg++;
    }

    // スタンプログ (氏名/大学名/属性のみ)
    var stmpSheet = ss.getSheetByName(SHEET.STAMP_LOG);
    if (stmpSheet && stmpSheet.getLastRow() > 1) {
      var d = stmpSheet.getDataRange().getValues();
      var h = d[0].map(String);
      var sidC = h.indexOf('studentId');
      for (var i = 1; i < d.length; i++) {
        if (String(d[i][sidC]) !== sid) continue;
        if (fields.name)     { var c = h.indexOf('氏名');   if (c>=0) stmpSheet.getRange(i+1,c+1).setValue(fields.name); }
        if (fields.school)   { var c = h.indexOf('大学名'); if (c>=0) stmpSheet.getRange(i+1,c+1).setValue(fields.school); }
        if (fields.category) { var c = h.indexOf('属性');   if (c>=0) stmpSheet.getRange(i+1,c+1).setValue(fields.category); }
        updated.stampLog++;
      }
    }

    // 景品交換ログ (氏名のみ)
    var prizeSheet = ss.getSheetByName(SHEET.PRIZE_LOG);
    if (prizeSheet && prizeSheet.getLastRow() > 1) {
      var d = prizeSheet.getDataRange().getValues();
      var h = d[0].map(String);
      var sidC = h.indexOf('studentId');
      for (var i = 1; i < d.length; i++) {
        if (String(d[i][sidC]) !== sid) continue;
        if (fields.name) { var c = h.indexOf('氏名'); if (c>=0) prizeSheet.getRange(i+1,c+1).setValue(fields.name); }
        updated.prizeLog++;
      }
    }
  });

  // 大学名・属性が変わったら studentId を採番し直し、全シートを追従させる
  var newSid = '';
  if (fields.school || fields.category) {
    var ssX;
    try { ssX = eventSs_(eventId); } catch (e3) { ssX = null; }
    if (ssX) {
      newSid = renumberStudentId_(ssX, eventId, sid);
    }
  }

  return ok_({ updated: updated, studentId: newSid || sid, renumbered: !!newSid });
  } finally {
    lock.releaseLock();
  }
}

// 学生マスターの現在の大学名・属性から studentId を組み直し、変わっていれば
// studentId を持つ全シートと CONFIG.womenPairings を追従させる。
// cardToken は触らない（名札QRを再発行しないため）。
// 戻り値: 新しい studentId（変更が不要だった場合は空文字）
function renumberStudentId_(ss, eventId, oldSid) {
  var stuSheet = ss.getSheetByName(SHEET.STUDENTS);
  if (!stuSheet || stuSheet.getLastRow() < 2) return '';
  var data    = stuSheet.getDataRange().getValues();
  var headers = data[0].map(String);
  var sidCol  = headers.indexOf('studentId');
  var schCol  = headers.indexOf('大学名');
  var attrCol = headers.indexOf('属性');
  if (sidCol < 0 || schCol < 0 || attrCol < 0) return '';

  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][sidCol]) === oldSid) { rowIdx = i; break; }
  }
  if (rowIdx < 0) return '';

  var school = String(data[rowIdx][schCol] || '').trim();
  var attr   = String(data[rowIdx][attrCol] || '').trim();
  if (!school) return '';

  // 大学コード（未登録の大学なら学生登録と同じく仮コードを自動採番する）
  var uni = loadUniversityMaster_();
  var uniCode = uni.uniMap.get(school);
  if (!uniCode) {
    uniCode = String(uni.maxAutoCode + 1).padStart(4, '0');
    saveNewUniversities_([[school, uniCode, '自動割当(要確認)']]);
  }

  var classCode  = PRE_CLASS_CODES[attr] || 'V';
  var yearMatch  = String(eventId).match(/(\d{4})/);
  var yearSuffix = yearMatch ? yearMatch[1].slice(-1) : String(new Date().getFullYear()).slice(-1);
  var idPrefix   = yearSuffix + uniCode + classCode;

  if (oldSid.substring(0, idPrefix.length) === idPrefix) return ''; // 大学も区分も実質変わっていない

  var newSid = idPrefix + String(nextStudentSeq_(data, sidCol, idPrefix)).padStart(2, '0');

  // studentId を持つ全シートを書き換える（studentId 列のみ・他列は触らない）
  [SHEET.STUDENTS, SHEET.STAMP_LOG, SHEET.STAMP_PARTICIPANTS,
   SHEET.PRIZE_LOG, SHEET.PRE_REG, SHEET.VIEW_LOG].forEach(function (name) {
    replaceStudentIdInSheet_(ss, name, oldSid, newSid);
  });
  replaceStudentIdInWomenPairings_(ss, oldSid, newSid);
  CacheService.getScriptCache().remove('cfg_' + eventId);
  return newSid;
}

// 指定シートの studentId 列だけを走査し、oldSid の行を newSid に差し替える
function replaceStudentIdInSheet_(ss, sheetName, oldSid, newSid) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet || sheet.getLastRow() < 2) return 0;
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(String);
  var sidCol  = headers.indexOf('studentId');
  if (sidCol < 0) return 0;
  var range = sheet.getRange(2, sidCol + 1, sheet.getLastRow() - 1, 1);
  var vals  = range.getValues();
  var n = 0;
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0]) === oldSid) { vals[i][0] = "'" + newSid; n++; }
  }
  if (n > 0) range.setValues(vals);
  return n;
}

// CONFIG の womenPairings 内の studentId を差し替える
function replaceStudentIdInWomenPairings_(ss, oldSid, newSid) {
  var sheet = ss.getSheetByName(SHEET.CONFIG);
  if (!sheet || sheet.getLastRow() < 1) return;
  var data = sheet.getDataRange().getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][0]) !== 'womenPairings') continue;
    var raw = String(data[i][1] || '');
    if (!raw) return;
    try {
      var arr = JSON.parse(raw);
      if (!Array.isArray(arr)) return;
      var changed = false;
      arr.forEach(function (pair) {
        if (!pair) return;
        ['a', 'b', 'c'].forEach(function (k) {
          if (String(pair[k] || '') === oldSid) { pair[k] = newSid; changed = true; }
        });
      });
      if (changed) sheet.getRange(i + 1, 2).setValue(JSON.stringify(arr));
    } catch (err) { /* 壊れたJSONは触らない */ }
    return;
  }
}

// イベントの基本情報（名前・開始日・終了日・状態）とCONFIG期間設定を更新

  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.eventId) return err_('missing_params');

  // EVENT_LIST を更新（getCurrentEvent の日付比較用）

  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const idCol   = headers.indexOf('eventId');

  if (rowIdx < 0) return err_('event_not_found');

  const sheetRow = rowIdx + 1;
  const setList = (colName, value) => {
    const col = headers.indexOf(colName);
    if (col >= 0 && value) sheet.getRange(sheetRow, col + 1).setValue(value);
  };
  if (p.eventName) setList('イベント名', p.eventName);
  if (p.startDate) setList('開始日', p.startDate);
  if (p.endDate)   setList('終了日', p.endDate);
  if (p.status)    setList('状態', p.status);

  // CONFIG シートも更新（スタンプ・交換期間）
  if (p.stampStartAt || p.stampEndAt || p.exchangeDeadline) {
    const cfg = eventSs_(p.eventId).getSheetByName(SHEET.CONFIG);
    const cfgData = cfg.getDataRange().getValues();
    const setConfig = (key, value) => {
      if (!value) return;
      const idx = cfgData.findIndex(r => String(r[0]) === key);
      if (idx >= 0) cfg.getRange(idx + 1, 2).setValue(value);
      else cfg.appendRow([key, value]);
    };
    if (p.stampStartAt)     setConfig('stampStartAt',     p.stampStartAt);
    if (p.stampEndAt)       setConfig('stampEndAt',       p.stampEndAt);
    if (p.exchangeDeadline) setConfig('exchangeDeadline', p.exchangeDeadline);
  }

  CacheService.getScriptCache().remove('cfg_' + String(p.eventId));
  return ok_({ updated: true });
}

// 統計（学生数・スタンプ数・参加者数・景品交換数）
function actionAdminGetStats_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event) return err_('missing_event');

  const ss = eventSs_(p.event);

  const studentSheet = ss.getSheetByName(SHEET.STUDENTS);
  let studentCount = 0, walkInCount = 0;
  let stuLunchSat = 0, stuLunchSun = 0;

  // 弁当希望(はい/いいえ)は「事前登録」シートにのみ記録される(STUDENTSには書き込まれない)。
  // studentId をキーに引けるようマップ化しておく。
  const preRegLunch = {};
  const preRegSheet = ss.getSheetByName(SHEET.PRE_REG);
  if (preRegSheet && preRegSheet.getLastRow() > 1) {
    const pData  = preRegSheet.getDataRange().getValues();
    const pHdrs  = pData[0].map(String);
    const sidCol = pHdrs.indexOf('studentId');
    const lsCol  = pHdrs.indexOf('弁当_土');
    const llCol  = pHdrs.indexOf('弁当_日');
    if (sidCol >= 0) {
      pData.slice(1).forEach(r => {
        preRegLunch[String(r[sidCol])] = {
          sat: lsCol >= 0 ? String(r[lsCol]) : '',
          sun: llCol >= 0 ? String(r[llCol]) : '',
        };
      });
    }
  }

  // 土日両日参加が前提のため、そもそも弁当希望を聞いていない区分(出場選手クラス)は
  // STUDENTSの属性(＝現在の正式な名簿。手動でのクラス変更・選手交代も反映済み)を基準に
  // 無条件で+1する。それ以外(補欠ドライバー/メカニック/応援学生)は事前登録の回答を見る。
  const AUTO_LUNCH_ATTRS = ['Aドライバー', 'Bドライバー', 'Cドライバー', '女子クラスドライバー'];

  if (studentSheet && studentSheet.getLastRow() > 1) {
    const sData    = studentSheet.getDataRange().getValues();
    const sHeaders = sData[0].map(String);
    const regCol   = sHeaders.indexOf('登録種別');
    const attrCol  = sHeaders.indexOf('属性');
    const sidCol   = sHeaders.indexOf('studentId');
    studentCount = sData.length - 1;
    walkInCount  = regCol >= 0
      ? sData.slice(1).filter(r => String(r[regCol]) === '当日').length
      : 0;

    sData.slice(1).forEach(r => {
      const attr = attrCol >= 0 ? String(r[attrCol]) : '';
      if (AUTO_LUNCH_ATTRS.indexOf(attr) >= 0) {
        stuLunchSat += 1;
        stuLunchSun += 1;
        return;
      }
      const lunch = sidCol >= 0 ? preRegLunch[String(r[sidCol])] : null;
      if (!lunch) return;
      stuLunchSat += lunch.sat === 'はい' ? 1 : (Number(lunch.sat) || 0);
      stuLunchSun += lunch.sun === 'はい' ? 1 : (Number(lunch.sun) || 0);
    });
  }

  // 企業側の食券（出展申込シート）
  const entrySheet = ss.getSheetByName(SHEET.COMPANY_ENTRIES);
  let coLunchSat = 0, coLunchSun = 0;
  if (entrySheet && entrySheet.getLastRow() > 1) {
    const eData    = entrySheet.getDataRange().getValues();
    const eHeaders = eData[0].map(String);
    const elsCol   = eHeaders.indexOf('昼食土');
    const ellCol   = eHeaders.indexOf('昼食日');
    if (elsCol >= 0 || ellCol >= 0) {
      eData.slice(1).forEach(r => {
        if (elsCol >= 0) coLunchSat += Number(r[elsCol]) || 0;
        if (ellCol >= 0) coLunchSun += Number(r[ellCol]) || 0;
      });
    }
  }

  const stampSheet       = ss.getSheetByName(SHEET.STAMP_LOG);
  const participantSheet = ss.getSheetByName(SHEET.STAMP_PARTICIPANTS);
  const prizeSheet       = ss.getSheetByName(SHEET.PRIZE_LOG);

  // スタンプ参加者（イベント内で1個以上スタンプした学生のユニーク数）
  let stampParticipants = 0;
  if (stampSheet && stampSheet.getLastRow() > 1) {
    const sRows = stampSheet.getRange(2, 1, stampSheet.getLastRow() - 1, 3).getValues();
    const uniq  = new Set(sRows.filter(r => String(r[1]) === String(p.event)).map(r => String(r[2])));
    stampParticipants = uniq.size;
  }

  // 景品交換済み個数（行数ではなく claimedCount の合計）
  let prizeCount = 0;
  if (prizeSheet && prizeSheet.getLastRow() > 1) {
    const pData    = prizeSheet.getDataRange().getValues();
    const pHeaders = pData[0].map(String);
    const pEvCol   = pHeaders.indexOf('eventId');
    const pCntCol  = pHeaders.indexOf('claimedCount');
    prizeCount = pData.slice(1)
      .filter(r => String(r[pEvCol]) === String(p.event))
      .reduce((sum, r) => sum + (pCntCol >= 0 ? Number(r[pCntCol]) || 1 : 1), 0);
  }

  return ok_({
    studentCount,
    preRegisteredCount: studentCount - walkInCount,
    walkInCount,
    stampParticipants,
    stampCount:       stampSheet       ? Math.max(0, stampSheet.getLastRow()       - 1) : 0,
    participantCount: participantSheet ? Math.max(0, participantSheet.getLastRow() - 1) : 0,
    prizeCount,
    lunch: {
      student: { sat: stuLunchSat, sun: stuLunchSun },
      company: { sat: coLunchSat,  sun: coLunchSun },
    },
  });
}

 *
 * 集計方針（2026-08-25 ユーザー確定）:
 *   - 選手（出場選手 FGクラス/女子クラスドライバー）: 来場予定日を申告する項目自体が
 *     フォームに無い（レースが2日間にまたがるため）。**土日とも「来場確定」として
 *     一律加算する**。来場実績（登録/スタンプ/QR）の集計には含めない。
 *   - 補欠ドライバー・見学応援・当日参加登録: 「選手以外」として、
 *     次の3方向のいずれかで存在すれば、その日の来場者としてカウントする
 *     （studentId で重複排除・和集合）。
 *       ① 登録   … STAMP_PARTICIPANTS（当日参加登録は登録と同時にここへ行が作られる）
 *       ② スタンプ … STAMP_LOG（ブースでのスタンプ取得）
 *       ③ QR読み取り … VIEW_LOG（企業が学生QRを読む＝名刺交換）
 *   - 「概算合計」は 選手（確定）+ 選手以外（来場実績） で、選手込みの数値であることを
 *     必ず明示する（呼び出し元の表示でも「※選手込み」等を付けること）。
 *
 * 併せて「活動記録」（スタンプ取得・QR読み取りの延べ回数と実人数）を
 * 選手／選手以外に分けて返す（2026-08-25 追加）。こちらは来場者数とは独立した
 * 行動量の集計で、選手も対象に含める（来場者数では一律加算のため実績を見ないが、
 * 選手がどれだけ活動したかは別途分析価値があるという運用判断）。
 *
 * ⚠ この関数は読み取りのみ。シートへの書き込みは一切ない。
 */
// 実出場するドライバーか（FGクラス＋女子クラス）。事前登録シートの「参加区分」列
// （フォームの送信値そのもの）を判定する。js/admin.js の同名関数と揃えてある
// （2026-08-25 発覚: この関数が定義されないまま actionAdminGetAttendanceReport_ から
// 呼ばれており、事前登録データがあるイベントで ReferenceError → server_error になっていた）。
function isCompetingDriverCategory_(category) {
  return category === '出場選手(FGクラスドライバー)'
      || category === '出場選手(女子クラスドライバー)';
}

// 学年を開催報告書の6区分（大学1〜4年生／大学院生／その他）に正規化する。
// ⚠ フォーム(app/register-pre.html の f-year)には「大学院生」の選択肢が無い。
//   大学院生は「その他」＋自由記述（例「大学院修士2年」）でしか捕捉できないため、
//   報告書でも自由記述を人が読んで数えていた。ここでも文字列で判定する。
// ⚠ 短期大学／専門学校／自動車大学校は報告書に対応する区分が無いので「その他」に寄せる。
//   （報告書 chart1 の その他=7 はこれらを含んでいたと考えられる）
function normalizeYear_(raw) {
  const s = String(raw == null ? '' : raw).trim();
  if (!s) return '未回答';
  if (/院|修士|博士/.test(s)) return '大学院生';
  const m = s.match(/^大学学部([1-4])年生$/);
  if (m) return '大学' + m[1] + '年生';
  return 'その他';
}

// 学部学科（自由入力）を開催報告書の3区分に分類する。
// ⚠ キーワードによる自動分類であり、正確性は保証されない。分類結果の内訳
//   （facultiesDetail）で元の文字列を返すので、報告書に使う前に必ず目視確認すること。
// ⚠ 理工判定に「科学」を入れると「社会科学部」を取り違えるため入れていない。
var FACULTY_SCI_RE_ = /理工|工学|工業|理学|情報|機械|電気|電子|建築|土木|物理|化学|数理|数学|生物|生命|農学|医学|薬学|看護|歯学|獣医|航空|海洋|環境/;
var FACULTY_HUM_RE_ = /経済|経営|商学|法学|文学|社会|人文|政治|外国語|国際|教育|心理|観光|福祉|言語|哲学|史学/;
function classifyFaculty_(raw) {
  const s = String(raw == null ? '' : raw).trim();
  if (!s) return '未回答';
  if (FACULTY_SCI_RE_.test(s)) return '理工学系';
  if (FACULTY_HUM_RE_.test(s)) return '人文・社会経済系';
  return 'その他';
}

function actionAdminGetAttendanceReport_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event) return err_('missing_event');

  const ss = eventSs_(p.event);
  const tz = Session.getScriptTimeZone();
  const dayOf = d => (d instanceof Date) ? Utilities.formatDate(d, tz, 'yyyy/MM/dd') : null;

  // イベントの開催日（2日分）を EVENT_LIST から取得し、それ以外の日付は無視する
  // （テストデータ等の日付混入で日を増やさないため）。
  const evSheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
  const evData  = evSheet.getDataRange().getValues();
  const evHdr   = evData[0].map(String);
  const evRow   = evData.slice(1).find(r => String(r[evHdr.indexOf('eventId')]) === String(p.event));
  if (!evRow) return err_('event_not_found');
  const startDate = new Date(evRow[evHdr.indexOf('開始日')]);
  const endDate   = new Date(evRow[evHdr.indexOf('終了日')]);
  const days = [];
  for (let d = new Date(startDate); d <= endDate; d.setDate(d.getDate() + 1)) {
    days.push(Utilities.formatDate(d, tz, 'yyyy/MM/dd'));
  }

  // ── 選手（確定・両日一律加算） ──────────────────────
  // driverIds は「選手以外」の集計から選手を除外するために使う。
  // これが無いと、選手がスタンプを押したりQRを読まれたりした場合に
  // 「選手」（一律加算）と「選手以外」（来場実績）の両方へ計上され、
  // 合計が実人数より多く出る（2026-08-25 修正。上のコメントにある
  // 「来場実績の集計には含めない」という設計意図に実装を合わせた）。
  // ⚠ 事前登録シートは弁当の集計でも使うので、読み込みは1回だけにして使い回す。
  const preSheet = ss.getSheetByName(SHEET.PRE_REG);
  const preData_ = (preSheet && preSheet.getLastRow() > 1) ? preSheet.getDataRange().getValues() : null;
  let driverCount = 0, fgDrivers = 0, womenDrivers = 0;
  const driverIds = new Set();
  if (!preSheet) {
    warnings.push('シート「' + SHEET.PRE_REG + '」が見つかりません。選手数・弁当の集計は0件になります。');
  }
  if (preData_) {
    const pData = preData_;
    const pHdr  = pData[0].map(String);
    const catCol = pHdr.indexOf('参加区分');
    const sidCol = pHdr.indexOf('studentId');
    // ⚠ 列が無いと「選手0人」という正常値に見えてしまうため必ず警告する。
    if (catCol < 0) warnings.push('シート「' + SHEET.PRE_REG + '」に「参加区分」列がありません。選手を判定できないため選手数は0になります。');
    if (sidCol < 0) warnings.push('シート「' + SHEET.PRE_REG + '」に「studentId」列がありません。選手と来場記録を突き合わせられません。');
    pData.slice(1).forEach(r => {
      const cat = String(r[catCol] || '');
      if (!isCompetingDriverCategory_(cat)) return;
      driverCount++;
      if (cat === '出場選手(女子クラスドライバー)') womenDrivers++; else fgDrivers++;
      if (sidCol >= 0) {
        const sid = String(r[sidCol] || '');
        if (sid) driverIds.add(sid);
      }
    });
  }

  // ── 選手以外（3方向の和集合・日別） ──────────────────
  const byDay = {}; // day -> Set(studentId)
  days.forEach(d => { byDay[d] = new Set(); });
  const bySource = {}; // day -> { registration, stamp, view } の件数（内訳表示用。重複除去前の延べ数）
  days.forEach(d => { bySource[d] = { registration: 0, stamp: 0, view: 0 }; });

  // 選手のうち、実際に何らかの記録（スタンプ開始・当日登録／スタンプ／QR読み取り）が
  // 残っている人の studentId を日別にためておく（2026-08-26 追加）。
  // 来場者数の集計には使わない（選手は一律加算のまま）。
  // ⚠ 選手は必ず来場している前提なので、ここで記録が無い＝「来場したが何も操作しなかった」。
  const driverRecordByDay = {};
  days.forEach(d => { driverRecordByDay[d] = new Set(); });

  // ⚠ ログシートは1回だけ読んで正規化配列にし、以降の集計で使い回す（2026-08-26 変更）。
  //   以前は STAMP_LOG / VIEW_LOG を来場判定・活動記録・ブース別/企業別で
  //   それぞれ読み直しており、1シートあたり3回 getDataRange していた。
  // ⚠ シートや必須列が欠けている場合、黙って0件にせず warnings に積んで呼び出し元へ返す。
  //   「本当に記録が無い」のか「集計できていない」のかを画面で区別するため。
  const warnings = [];
  const readLog_ = (sheetName, extraCols) => {
    const out = { rows: [], ok: false };
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) { warnings.push('シート「' + sheetName + '」が見つかりません。関連する集計は0件として表示されます。'); return out; }
    out.ok = true;
    if (sheet.getLastRow() < 2) return out;   // 行が無いのは正常（記録なし）
    const data = sheet.getDataRange().getValues();
    const hdr  = data[0].map(String);
    const tCol = hdr.indexOf('時刻');
    const sCol = hdr.indexOf('studentId');
    if (tCol < 0 || sCol < 0) {
      out.ok = false;
      warnings.push('シート「' + sheetName + '」に「時刻」または「studentId」列がありません。関連する集計は0件として表示されます。');
      return out;
    }
    const exIdx = (extraCols || []).map(n => {
      const i = hdr.indexOf(n);
      if (i < 0) warnings.push('シート「' + sheetName + '」に「' + n + '」列がありません。この列を使う集計は空欄になります。');
      return i;
    });
    data.slice(1).forEach(r => {
      const day = dayOf(r[tCol]);
      if (!day || !byDay[day]) return;        // 開催日外は無視
      const sid = String(r[sCol] || '');
      if (!sid) return;
      const row = { day: day, sid: sid };
      (extraCols || []).forEach((n, i) => {
        row[n] = exIdx[i] >= 0 ? String(r[exIdx[i]] || '').trim() : '';
      });
      out.rows.push(row);
    });
    return out;
  };
  const participantsLog_ = readLog_(SHEET.STAMP_PARTICIPANTS);
  const stampLog_        = readLog_(SHEET.STAMP_LOG, ['企業名']);
  const viewLog_         = readLog_(SHEET.VIEW_LOG,  ['companyId']);

  const addRows = (log, key) => {
    log.rows.forEach(r => {
      // 選手は一律加算済みなので来場者数には入れない（二重計上を防ぐ）。
      // ただし記録が残っていること自体は控えておく。
      if (driverIds.has(r.sid)) { driverRecordByDay[r.day].add(r.sid); return; }
      byDay[r.day].add(r.sid);
      bySource[r.day][key]++;
    });
  };
  addRows(participantsLog_, 'registration');
  addRows(stampLog_,        'stamp');
  addRows(viewLog_,         'view');

  // ⚠ 選手（driverCount）は上の3方向の集計に混ざらないよう、対象シートには
  //   selectしていない点に注意。同一学生が選手かつ何らかの理由でスタンプ等に
  //   記録されていても、選手側と選手以外側で二重にならないよう分離してある。
  //   ただし選手が実際にスタンプ等の行を持つケース自体は、この関数では
  //   選手以外の集計にも入りうる（例: 出走選手が余興でスタンプを押した場合）。
  //   その場合は selectin 上「選手」と「選手以外」の両方に1名ずつ計上され、
  //   合計が実人数より多く出うる。概算値としての制約として許容する
  //   （2026-08-25 ユーザー確認: 選手は来場実績の集計対象にしない方針のため、
  //   これ以上厳密にする場合は選手のstudentIdを選手以外の集計から除外する
  //   実装が必要。今回は「概算でよい」との要望のため見送り）。

  const byDayResult = {};
  days.forEach(d => {
    const nonDriver = byDay[d].size;
    byDayResult[d] = {
      drivers: driverCount,
      nonDrivers: nonDriver,
      total: driverCount + nonDriver,
      sources: bySource[d],
    };
  });

  // ── 活動記録（スタンプ取得・QR読み取り） ────────────
  // 来場者数の集計とは独立した「行動量」の集計（2026-08-25 追加）。
  // 来場者数では選手を一律加算にして実績から除外しているが、
  // 「選手が実際に何回スタンプを押したか／何回QRを読まれたか」は
  // 運用上の分析価値があるため、選手／選手以外を分けて延べ回数と実人数を出す。
  // ⚠ ここでの区分は driverIds（＝事前登録の参加区分が出場選手）による。
  //   補欠ドライバーは選手以外に入る（isCompetingDriverCategory_ が false のため）。
  // perSid を渡すと、学生ごとの件数も同じ走査で数える。
  const countActivity_ = (log, perSid) => {
    const res = {
      driver:    { count: 0, users: new Set() },
      nonDriver: { count: 0, users: new Set() },
    };
    log.rows.forEach(r => {
      if (perSid) perSid[r.sid] = (perSid[r.sid] || 0) + 1;
      const bucket = driverIds.has(r.sid) ? res.driver : res.nonDriver;
      bucket.count++;
      bucket.users.add(r.sid);
    });
    return res;
  };
  const stampBySid_ = {};   // 学生ごとのスタンプ取得回数（大学別集計・CSV用）
  const stampAct_ = countActivity_(stampLog_, stampBySid_);
  const viewAct_  = countActivity_(viewLog_);
  const actOut_   = a => ({ count: a.count, users: a.users.size });

  // ── 学生属性（開催報告書の Student Attribute 相当） ──
  // docs/FG2026富士1報告書圧縮版.pdf の構成に合わせ、学年・都道府県・学部学科を
  // 選手／応援に分けて集計する（2026-08-25 追加）。
  // ⚠ 対象は「選手」＋「来場記録のある選手以外」のみ。学生マスターに登録だけあって
  //   来場記録が無い学生は、来場者数の集計と同じ理由で除外する。
  // ⚠ 学部学科はフリーテキストのため、報告書にある「理工学系／人文・社会経済系」の
  //   分類は自動化していない。生の学部学科名の件数を返すので分類は手作業で行うこと。
  const attendedNonDrivers = new Set();
  days.forEach(d => byDay[d].forEach(sid => attendedNonDrivers.add(sid)));

  // years/faculties は報告書の区分に正規化した値。yearsRaw/facultiesDetail は
  // 正規化前の生データで、分類が妥当か目視確認するために返している。
  const mkDist_  = () => ({
    years: {}, yearsRaw: {}, prefectures: {},
    faculties: {}, facultiesDetail: { '理工学系': {}, '人文・社会経済系': {}, 'その他': {}, '未回答': {} },
  });
  const attrs_   = { driver: mkDist_(), nonDriver: mkDist_() };
  const bump_    = (obj, v) => { const k = String(v == null ? '' : v).trim() || '未回答'; obj[k] = (obj[k] || 0) + 1; };
  const schoolSet = new Set();
  // ⚠ 学生マスターは属性集計・無記録者・学生ごとの明細で使うため、読み込みは1回だけ。
  const studentSheet2 = ss.getSheetByName(SHEET.STUDENTS);
  const studentData_  = (studentSheet2 && studentSheet2.getLastRow() > 1)
    ? studentSheet2.getDataRange().getValues() : null;
  if (!studentSheet2) {
    warnings.push('シート「' + SHEET.STUDENTS + '」が見つかりません。学生属性・大学別集計は空になります。');
  } else if (studentData_ && studentData_[0].map(String).indexOf('studentId') < 0) {
    warnings.push('シート「' + SHEET.STUDENTS + '」に「studentId」列がありません。学生属性・大学別集計は空になります。');
  }
  if (studentData_) {
    const sData = studentData_;
    const sHdr  = sData[0].map(String);
    const sidC  = sHdr.indexOf('studentId');
    const yrC   = sHdr.indexOf('学年');
    const prefC = sHdr.indexOf('住所(都道府県)');
    const facC  = sHdr.indexOf('学部学科');
    const schC  = sHdr.indexOf('大学名');
    sData.slice(1).forEach(r => {
      const sid = sidC >= 0 ? String(r[sidC] || '') : '';
      if (!sid) return;
      const isDriver = driverIds.has(sid);
      if (!isDriver && !attendedNonDrivers.has(sid)) return; // 来場記録なしは除外
      const t = isDriver ? attrs_.driver : attrs_.nonDriver;
      if (yrC   >= 0) { bump_(t.years, normalizeYear_(r[yrC])); bump_(t.yearsRaw, r[yrC]); }
      if (prefC >= 0) bump_(t.prefectures, r[prefC]);
      if (facC  >= 0) {
        const cls = classifyFaculty_(r[facC]);
        bump_(t.faculties, cls);
        bump_(t.facultiesDetail[cls], r[facC]);
      }
      if (schC  >= 0) { const sc = String(r[schC] || '').trim(); if (sc) schoolSet.add(sc); }
    });
  }

  // ── ブース別スタンプ数（スタンプラリー結果） ──────────
  // docs/スタンプラリー参加学生集計結果.xlsm と同じ切り口:
  // 日別 × 企業ブース別 × 選手／応援 のスタンプ取得数（2026-08-25 追加）。
  const boothMap_ = {};
  stampLog_.rows.forEach(r => {
    const booth = r['企業名'] || '(企業名なし)';
    if (!boothMap_[booth]) {
      boothMap_[booth] = {};
      days.forEach(d => { boothMap_[booth][d] = { driver: 0, nonDriver: 0 }; });
    }
    const cell = boothMap_[booth][r.day];
    if (driverIds.has(r.sid)) cell.driver++; else cell.nonDriver++;
  });
  const boothRows_ = Object.keys(boothMap_).map(name => {
    const perDay = days.map(d => {
      const c = boothMap_[name][d];
      return { driver: c.driver, nonDriver: c.nonDriver, total: c.driver + c.nonDriver };
    });
    return { name, days: perDay, total: perDay.reduce((s, c) => s + c.total, 0) };
  }).sort((a, b) => String(a.name).localeCompare(String(b.name), 'ja')); // 名前順（件数順はランキング表示になるため使わない）

  // 出場校エントリー数（＝報告書の「参加校数」）／出展企業数（＝ブース出展数）
  const schoolSheet2 = ss.getSheetByName(SHEET.SCHOOL_ENTRIES);
  const schoolEntryCount = (schoolSheet2 && schoolSheet2.getLastRow() > 1) ? schoolSheet2.getLastRow() - 1 : 0;
  // ⚠ 企業マスターの全行ではなく「出店中」の企業だけを数える（2026-08-26 修正）。
  //   管理画面の企業一覧にある「ブース出店」トグル（出店中／出店なし）が stampRally 列。
  //   判定は他の参照箇所（actionGetEventCompanies_ 等）と同じく
  //   「明示的に 'FALSE' のときだけ出店なし」。列が無い・空欄なら出店中として扱う
  //   （stampRally 列を持たない既存イベントとの後方互換）。
  const coSheet2 = ss.getSheetByName(SHEET.COMPANIES);
  if (!coSheet2) warnings.push('シート「' + SHEET.COMPANIES + '」が見つかりません。出展ブース数は0になります。');
  let companyCount = 0;
  const companyNames_ = {};   // companyId -> 企業名（QR読み取り集計の表示用）
  if (coSheet2 && coSheet2.getLastRow() > 1) {
    const cData    = coSheet2.getDataRange().getValues();
    const cHeaders = cData[0].map(String);
    const cIdCol   = cHeaders.indexOf('companyId');
    const cNameCol = cHeaders.indexOf('企業名');
    const cRallyCol = cHeaders.indexOf('stampRally');
    cData.slice(1).forEach(r => {
      const cid = cIdCol >= 0 ? String(r[cIdCol] || '').trim() : '';
      if (cIdCol >= 0 && !cid) return;                                     // 空行は数えない
      if (cid && cNameCol >= 0) companyNames_[cid] = String(r[cNameCol] || '').trim();
      if (cRallyCol >= 0 && String(r[cRallyCol]).toUpperCase() === 'FALSE') return; // 出店なし
      companyCount++;
    });
  }

  // ── 企業別のQR読み取り ──────────────────────────────
  // 「その企業が読み取った学生の実人数」。企業×学生の組で重複排除して数える。
  // ⚠ 行数を数えてはいけない（2026-08-26 修正）。actionSaveViewLog_ の重複抑制は
  //   source='company_auto' のときだけで、手動メール登録(manual_email)には無い。
  //   さらに company_auto でも「読んでから追記する」までロックを取らないため、
  //   同時実行では重複行が残りうる。行数のままだと実人数より多く出て、
  //   対外報告に使う値が膨らむ。ここで明示的に一意化しておく。
  // ⚠ 日別には割らない。company_auto は再読み取り時に時刻だけ更新するため、
  //   日で割ると「最後に読まれた日」に寄って実態と合わない。イベント通算で返す。
  const viewPairs_ = {};   // companyId -> { driver:Set, nonDriver:Set }
  viewLog_.rows.forEach(r => {
    const cid = r['companyId'];
    if (!cid) return;
    if (!viewPairs_[cid]) viewPairs_[cid] = { driver: new Set(), nonDriver: new Set() };
    if (driverIds.has(r.sid)) viewPairs_[cid].driver.add(r.sid);
    else viewPairs_[cid].nonDriver.add(r.sid);
  });
  const viewByCompany_ = {};
  Object.keys(viewPairs_).forEach(cid => {
    viewByCompany_[cid] = {
      driver:    viewPairs_[cid].driver.size,
      nonDriver: viewPairs_[cid].nonDriver.size,
    };
  });

  // 学生ごとの「接点企業数」。QR閲覧ログは同一企業×同一学生の行が複数残りうるため、
  // 行数ではなく企業を一意化して数える（2026-08-26 修正。企業別QRと定義を揃える）。
  const viewCompaniesBySid_ = {};
  viewLog_.rows.forEach(r => {
    const cid = r['companyId'];
    if (!cid) return;
    if (!viewCompaniesBySid_[r.sid]) viewCompaniesBySid_[r.sid] = new Set();
    viewCompaniesBySid_[r.sid].add(cid);
  });
  const viewCompanyRows_ = Object.keys(viewByCompany_).map(cid => {
    const c = viewByCompany_[cid];
    return {
      companyId: cid,
      name: companyNames_[cid] || cid,
      driver: c.driver,
      nonDriver: c.nonDriver,
      total: c.driver + c.nonDriver,
    };
  }).sort((a, b) => String(a.name).localeCompare(String(b.name), 'ja')); // 名前順

  // ── 記録が無い学生・弁当の未消化 ──────────────────────
  // ⚠ 対象は選手以外のみ。選手は弁当希望を聞かず無条件で加算される運用
  //   （actionAdminGetStats_ の AUTO_LUNCH_ATTRS）かつ来場も一律扱いのため、
  //   「弁当ありなのに来ていない」という指標が成立しない。
  // 弁当_土 / 弁当_日 は曜日で対応付ける（開催日が土日固定でない場合に備え、
  // 日付の曜日を見て割り当てる。該当曜日が無い日は弁当の判定対象外）。
  const lunchColOfDay_ = day => {
    const m = String(day).match(/^(\d{4})\/(\d{2})\/(\d{2})$/);
    if (!m) return null;
    const w = new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])).getDay();
    if (w === 6) return '弁当_土';
    if (w === 0) return '弁当_日';
    return null;
  };
  // 弁当希望(はい/いいえ) × その日のアクション有無 のクロス集計（2026-08-26 変更）。
  //   lunchYesIdle … 弁当ありなのに無操作 ＝ 欠席、または来場したが何も操作しなかった人
  //   lunchNoActive … 弁当なしなのに来場して操作している ＝ 申込時の案内を読んでいない可能性
  // ⚠ 母集団は「事前登録シートにいる選手以外」。当日登録者は弁当の設問自体を通っていないため
  //   含めない（含めると lunchNoActive が当日登録者で埋まって意味をなさなくなる）。
  const lunchRows_ = [];
  if (preData_) {
    const pHdr   = preData_[0].map(String);
    const sidCol = pHdr.indexOf('studentId');
    days.forEach(day => {
      const colName = lunchColOfDay_(day);
      const col = colName ? pHdr.indexOf(colName) : -1;
      if (col < 0 || sidCol < 0) return;
      let yesActive = 0, yesIdle = 0, noActive = 0, noIdle = 0;
      preData_.slice(1).forEach(r => {
        const sid = String(r[sidCol] || '');
        if (!sid || driverIds.has(sid)) return;              // 選手は別枠で数える
        const wantsLunch = String(r[col]).trim() === 'はい';
        const acted      = byDay[day].has(sid);              // その日の記録の有無
        if (wantsLunch && acted)  yesActive++;
        else if (wantsLunch)      yesIdle++;
        else if (acted)           noActive++;
        else                      noIdle++;
      });
      lunchRows_.push({
        day: day, column: colName,
        yesActive: yesActive, yesIdle: yesIdle,
        noActive: noActive,   noIdle: noIdle,
        requested: yesActive + yesIdle,   // 弁当を希望した人数
        absent:    yesIdle,               // うち無操作（従来の「うち来場記録なし」）
      });
    });
  }

  // 選手の日別アクション有無。選手は必ず来場している前提なので、
  // idle は「来場したが何も操作しなかった人数」として読める。
  const driverDayRows_ = days.map(day => ({
    day:    day,
    active: driverRecordByDay[day].size,
    idle:   Math.max(0, driverIds.size - driverRecordByDay[day].size),
  }));

  // 登録はあるが、全開催日を通じて記録が一切ない学生（＝来場者数に計上されない人数）。
  // ⚠ 上の「弁当の未消化」は日別（その日に記録が無い）なのに対し、こちらは
  //   全開催日を通じて1件も記録が無い人だけを数える。片方の日しか来ていない学生は
  //   弁当側では未消化に入るが、こちらには入らない。数字がずれるのはこのため。
  let registeredStudents = 0, noRecordStudents = 0;
  if (studentData_) {
    const sidCol = studentData_[0].map(String).indexOf('studentId');
    if (sidCol >= 0) {
      studentData_.slice(1).forEach(r => {
        const sid = String(r[sidCol] || '').trim();
        if (!sid) return;
        registeredStudents++;
        if (driverIds.has(sid)) return;                      // 選手は別枠で数える
        if (!attendedNonDrivers.has(sid)) noRecordStudents++;
      });
    }
  }

  // ── 学生ごとのアクション明細（大学別集計とCSV出力の元データ） ──
  // 2026-08-26 追加。判断材料として出すだけで、この値による評価はしない。
  // ⚠ 「アクションあり」の定義は来場者数の集計と同じ（スタンプ開始・当日登録／
  //   スタンプ取得／企業によるQR読み取りのいずれかの記録が、その日に存在する）。
  // ⚠ 当日登録は登録と同時にスタンプ参加者行が作られるため、当日登録者は
  //   登録した日が必ず「アクションあり」になる。他の日は他の記録次第。
  // ⚠ 会場に来たが何も操作しなかった人と、そもそも来ていない人は区別できない。
  const lunchBySid_ = {};
  if (preData_) {
    const pHdr = preData_[0].map(String);
    const sidC = pHdr.indexOf('studentId');
    const satC = pHdr.indexOf('弁当_土');
    const sunC = pHdr.indexOf('弁当_日');
    if (sidC >= 0) {
      preData_.slice(1).forEach(r => {
        const sid = String(r[sidC] || '').trim();
        if (!sid) return;
        lunchBySid_[sid] = {
          sat: satC >= 0 ? String(r[satC]).trim() : '',
          sun: sunC >= 0 ? String(r[sunC]).trim() : '',
        };
      });
    }
  }

  const studentRows_ = [];
  if (studentData_) {
    const sHdr   = studentData_[0].map(String);
    const gi     = name => sHdr.indexOf(name);
    const sidCol = gi('studentId');
    const nameCol = gi('氏名');
    const schCol  = gi('大学名');
    const yrCol   = gi('学年');
    const attrCol = gi('属性');
    const regCol  = gi('登録種別');
    if (sidCol >= 0) {
      studentData_.slice(1).forEach(r => {
        const sid = String(r[sidCol] || '').trim();
        if (!sid) return;
        const isDriver = driverIds.has(sid);
        const perDay = days.map(day => ({
          day: day,
          acted: isDriver ? driverRecordByDay[day].has(sid) : byDay[day].has(sid),
        }));
        const lunch = lunchBySid_[sid] || { sat: '', sun: '' };
        studentRows_.push({
          studentId: sid,
          name:      nameCol >= 0 ? String(r[nameCol] || '') : '',
          school:    schCol  >= 0 ? String(r[schCol]  || '') : '',
          year:      yrCol   >= 0 ? String(r[yrCol]   || '') : '',
          attribute: attrCol >= 0 ? String(r[attrCol] || '') : '',
          regType:   regCol  >= 0 ? String(r[regCol]  || '') : '',
          isDriver:  isDriver,
          days:      perDay,
          actedAny:  perDay.some(d => d.acted),
          stamps:    stampBySid_[sid] || 0,   // スタンプ取得の回数（1回ごとに別の記録なので行数でよい）
          views:     viewCompaniesBySid_[sid] ? viewCompaniesBySid_[sid].size : 0, // 接点企業数（企業で重複排除）
          lunchSat:  lunch.sat,
          lunchSun:  lunch.sun,
        });
      });
    }
  }

  // 選手のうち、開催日中に何の記録も残さなかった人数（2026-08-26 追加）。
  // 来場者数では選手を一律加算しているため、この人数は来場者数には影響しない。
  // 分母は driverIds（＝studentId を持つ選手）。事前登録行に studentId が無い選手は
  // 突き合わせようがないため、driverCount ではなく driverIds.size を分母にしている。
  const driverIdCount = driverIds.size;
  const driversAllDays = new Set();
  days.forEach(d => driverRecordByDay[d].forEach(sid => driversAllDays.add(sid)));
  const driversWithRecord = driversAllDays.size;
  const driversNoRecord   = Math.max(0, driverIdCount - driversWithRecord);

  // ── 景品（交換済み個数・交換回数） ──────────────────
  const prizeSheet = ss.getSheetByName(SHEET.PRIZE_LOG);
  let prizeItemCount = 0, prizeExchangeCount = 0;
  if (prizeSheet && prizeSheet.getLastRow() > 1) {
    const pData    = prizeSheet.getDataRange().getValues();
    const pHeaders = pData[0].map(String);
    const pEvCol   = pHeaders.indexOf('eventId');
    const pCntCol  = pHeaders.indexOf('claimedCount');
    const pRows    = pData.slice(1).filter(r => String(r[pEvCol]) === String(p.event));
    prizeExchangeCount = pRows.length;
    prizeItemCount = pRows.reduce((sum, r) => sum + (pCntCol >= 0 ? Number(r[pCntCol]) || 1 : 1), 0);
  }

  return ok_({
    eventId: p.event,
    days: days,
    drivers: driverCount,
    byDay: byDayResult,
    activity: {
      stamp: { driver: actOut_(stampAct_.driver), nonDriver: actOut_(stampAct_.nonDriver) },
      view:  { driver: actOut_(viewAct_.driver),  nonDriver: actOut_(viewAct_.nonDriver) },
    },
    summary: {
      dayCount:           days.length,
      schoolEntryCount:   schoolEntryCount,
      attendeeSchoolCount: schoolSet.size,
      fgDrivers:          fgDrivers,
      womenDrivers:       womenDrivers,
      companyCount:       companyCount,
    },
    attributes: attrs_,
    booth: { days: days, rows: boothRows_ },
    viewByCompany: { rows: viewCompanyRows_ },
    ops: {
      registeredStudents: registeredStudents,
      noRecordStudents:   noRecordStudents,
      driverIdCount:      driverIdCount,
      driversWithRecord:  driversWithRecord,
      driversNoRecord:    driversNoRecord,
      lunch:              lunchRows_,
      driverDays:         driverDayRows_,
    },
    students: studentRows_,
    warnings: warnings,
    prizeItemCount,
    prizeExchangeCount,
  });
}

// 企業を追加
function actionAdminAddCompany_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event) return err_('missing_event');
  if (!p.name)  return err_('missing_params');

  const sheet   = eventSs_(p.event).getSheetByName(SHEET.COMPANIES);
  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const idCol   = headers.indexOf('companyId');

  // companyId: 指定がなければ短いランダムIDを生成
  const companyId = String(p.companyId || '').trim()
    || ('co' + Utilities.getUuid().replace(/-/g, '').slice(0, 6));

  // 重複チェック
  if (data.slice(1).some(r => String(r[idCol]) === companyId)) {
    return err_('company_id_exists');
  }

  // 表示順: 既存行数 + 1
  const displayOrder = data.length; // ヘッダ含む行数 = 既存企業数 + 1
  // schema: companyId, 企業名, stampKey, viewKey, 表示順, hasNfc[, logoUrl]
  // logoUrl 列は header 駆動で後から読むため、ここでは付与しない（後から
  // adminUpdateCompany が列を作成・設定する。新規イベントは createEvent_ が列を持つ）
  sheet.appendRow([companyId, String(p.name), '', '', displayOrder, false]);
  return ok_({ companyId, name: String(p.name) });
}

// 企業フィールドの更新（logoUrl / stampRally）列が無ければ末尾に自動作成
function actionAdminUpdateCompany_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event || !p.companyId) return err_('missing_params');

  const ALLOWED = ['logoUrl', 'stampRally'];
  const updates = {};
  ALLOWED.forEach(k => { if (p[k] !== undefined) updates[k] = p[k]; });
  if (!Object.keys(updates).length) return err_('missing_params');

  const sheet   = eventSs_(p.event).getSheetByName(SHEET.COMPANIES);
  const data    = sheet.getDataRange().getValues();
  let   headers = data[0].map(String);
  const idCol   = headers.indexOf('companyId');

  const rowIdx = data.slice(1).findIndex(r => String(r[idCol]) === String(p.companyId));
  if (rowIdx < 0) return err_('not_found');

  Object.entries(updates).forEach(([key, val]) => {
    let col = headers.indexOf(key);
    if (col < 0) {
      sheet.getRange(1, headers.length + 1).setValue(key);
      col = headers.length;
      headers.push(key);
    }
    sheet.getRange(rowIdx + 2, col + 1).setValue(String(val));
  });

  CacheService.getScriptCache().remove('cfg_' + p.event);
  return ok_({ companyId: p.companyId, ...updates });
}

// 企業ロゴを Google Drive にアップロードして thumbnail URL を返す
function actionAdminUploadCompanyLogo_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event || !p.companyId || !p.base64) return err_('missing_params');

  // Step1: Drive フォルダ確保
  let folder;
  try {
    const root = getOrCreateChildFolder_(DriveApp.getRootFolder(), 'FGロゴ');
    folder     = getOrCreateChildFolder_(root, String(p.event));
  } catch (er) {
    console.error('logo_step1_folder:', er);
    return { ok: false, error: 'logo_step1_folder_failed', message: 'Step1(Driveフォルダ): ' + (er && er.message ? er.message : String(er)) };
  }

  // Step2: 既存ファイル削除
  // ⚠ 保存時のファイル名は「companyId.拡張子」なので、companyId そのものでは一致しない。
  //   以前は getFilesByName(companyId) で探していたため一度も削除されず、
  //   再アップロードのたびに公開設定のファイルがDriveへ蓄積していた。
  //   拡張子が変わる場合（png→jpg 等）もあるため、フォルダ内を走査して前方一致で消す。
  try {
    const prefix = String(p.companyId) + '.';
    const it = folder.getFiles();
    while (it.hasNext()) {
      const f = it.next();
      const nm = f.getName();
      if (nm === String(p.companyId) || nm.indexOf(prefix) === 0) f.setTrashed(true);
    }
  } catch (er) {
    console.error('logo_step2_trash:', er);
    // 削除失敗は致命的でないので続行
  }

  // Step3: ファイル作成
  let file;
  try {
    const mime  = p.mimeType || 'image/png';
    const ext   = mime.split('/')[1] || 'png';
    const bytes = Utilities.base64Decode(p.base64);
    const blob  = Utilities.newBlob(bytes, mime, String(p.companyId) + '.' + ext);
    file = folder.createFile(blob);
  } catch (er) {
    console.error('logo_step3_create:', er);
    return { ok: false, error: 'logo_step3_create_failed', message: 'Step3(ファイル作成): ' + (er && er.message ? er.message : String(er)) };
  }

  // Step4: 公開設定
  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (er) {
    console.error('logo_step4_sharing:', er);
    return { ok: false, error: 'logo_step4_sharing_failed', message: 'Step4(公開設定): ' + (er && er.message ? er.message : String(er)) };
  }

  const fileId = file.getId();
  const url    = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=s200';

  // Step5: シートの logoUrl 更新
  try {
    const sheet   = eventSs_(p.event).getSheetByName(SHEET.COMPANIES);
    const data    = sheet.getDataRange().getValues();
    let   headers = data[0].map(String);
    const idCol   = headers.indexOf('companyId');
    const rowIdx  = data.slice(1).findIndex(r => String(r[idCol]) === String(p.companyId));
    if (rowIdx >= 0) {
      let col = headers.indexOf('logoUrl');
      if (col < 0) {
        sheet.getRange(1, headers.length + 1).setValue('logoUrl');
        col = headers.length;
      }
      sheet.getRange(rowIdx + 2, col + 1).setValue(url);
      CacheService.getScriptCache().remove('cfg_' + p.event);
    }
  } catch (er) {
    console.error('logo_step5_sheet:', er);

// ============================================================
// 管理者パスワード復旧（メール宛リセット）
// ロックアウト時用の公開フロー。認証は「事前設定した復旧用メール受信」で担保。
//   1) adminRequestReset … 復旧用メールへ再設定コードを送信
//   2) adminConfirmReset  … コード＋新パスワードで ADMIN_KEY を再設定
// 最終フォールバックとして resetAdminKey()（エディタ実行）も用意。
// ============================================================

var ADMIN_RESET_TTL_MS_    = 15 * 60 * 1000;  // コード有効期限: 15分
var ADMIN_RESET_RESEND_MS_ = 60 * 1000;       // 再送制限: 60秒
var ADMIN_RESET_MAX_TRIES_ = 5;               // 検証失敗の上限

// 暗号学的乱数(UUID)由来の8文字コード（紛らわしい文字を避けるため16進大文字）
function genResetCode_() {
  return Utilities.getUuid().replace(/-/g, '').slice(0, 8).toUpperCase();
}

function sha256Hex_(s) {
  var bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256, String(s), Utilities.Charset.UTF_8);
  return bytes.map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}

function clearAdminReset_() {
  var props = PropertiesService.getScriptProperties();
  ['ADMIN_RESET_HASH', 'ADMIN_RESET_EXPIRES', 'ADMIN_RESET_ATTEMPTS', 'ADMIN_RESET_SENT_AT']
    .forEach(function (k) { props.deleteProperty(k); });
}

// 公開: 復旧用メール宛に再設定コードを送信
function actionAdminRequestReset_(p) {
  var props = PropertiesService.getScriptProperties();
  var email = props.getProperty('ADMIN_RECOVERY_EMAIL');
  // 復旧用メール未設定でも、設定有無を外部に漏らさないため一律 ok を返す
  if (!email) return ok_({ requested: true });
  // 再送レート制限（連続送信によるメール爆撃を防ぐ）
  var lastSent = Number(props.getProperty('ADMIN_RESET_SENT_AT') || 0);
  if (Date.now() - lastSent < ADMIN_RESET_RESEND_MS_) return ok_({ requested: true });

  var code = genResetCode_();
  props.setProperty('ADMIN_RESET_HASH', sha256Hex_(code));   // 平文は保存しない
  props.setProperty('ADMIN_RESET_EXPIRES', String(Date.now() + ADMIN_RESET_TTL_MS_));
  props.setProperty('ADMIN_RESET_ATTEMPTS', '0');
  props.setProperty('ADMIN_RESET_SENT_AT', String(Date.now()));

  try {
    MailApp.sendEmail({
      to: String(email),
      subject: '【FG管理パネル】パスワード再設定コード',
      body: [
        '管理パネルのパスワード再設定がリクエストされました。',
        '',
        '再設定コード: ' + code,
        '（有効期限: 15分・1回限り有効）',
        '',
        '管理パネルのログイン画面で、このコードと新しいパスワードを入力してください。',
        'お心当たりがない場合は、このメールを破棄してください。',
        '',
        '── FORMULA GYMKHANA 事務局システム',
      ].join('\n'),
    });
  } catch (err) {
    console.error('actionAdminRequestReset_ mail error:', err);
  }
  return ok_({ requested: true });
}

// 公開: コード＋新パスワードで ADMIN_KEY を再設定
function actionAdminConfirmReset_(p) {
  if (!p.code || !p.newKey) return err_('missing_params');
  var props = PropertiesService.getScriptProperties();
  var hash  = props.getProperty('ADMIN_RESET_HASH');
  var exp   = Number(props.getProperty('ADMIN_RESET_EXPIRES') || 0);
  var tries = Number(props.getProperty('ADMIN_RESET_ATTEMPTS') || 0);

  if (!hash || !exp)            return err_('reset_not_requested');
  if (Date.now() > exp)         { clearAdminReset_(); return err_('reset_expired'); }
  if (tries >= ADMIN_RESET_MAX_TRIES_) { clearAdminReset_(); return err_('reset_locked'); }

  if (sha256Hex_(String(p.code).trim().toUpperCase()) !== hash) {
    props.setProperty('ADMIN_RESET_ATTEMPTS', String(tries + 1));
    return err_('reset_invalid_code');
  }

  props.setProperty('ADMIN_KEY', String(p.newKey).trim());
  clearAdminReset_();
  return ok_({ reset: true });
}

// ── GASエディタから直接実行する保守用関数（最終フォールバック） ──
// Apps Scriptエディタで関数を選んで実行する。GASを所有するGoogleアカウントでのみ
// 実行可能なため、パネルから締め出されても確実に復旧できる。
// 使い方: 関数 resetAdminKey を選び実行 → 実行ログに新しいキーが出る。
//        引数付きで呼ぶには一時的に resetAdminKey('好きなキー') と書いて実行。
function resetAdminKey(newKey) {
  var key = (newKey && String(newKey).trim()) || (genResetCode_() + genResetCode_());
  PropertiesService.getScriptProperties().setProperty('ADMIN_KEY', String(key));
  clearAdminReset_();
  Logger.log('ADMIN_KEY を再設定しました。新しいキー: ' + key);
  return key;
}

// 前回イベントから企業を引き継ぎ（stampKey/viewKey/logoUrl を保持）
function actionAdminImportCompanies_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event || !p.sourceEvent) return err_('missing_params');
  if (String(p.event) === String(p.sourceEvent)) return err_('same_event');

  const sourceSheet = eventSs_(p.sourceEvent).getSheetByName(SHEET.COMPANIES);
  if (!sourceSheet || sourceSheet.getLastRow() < 2) return err_('no_source_companies');

  const targetSheet  = eventSs_(p.event).getSheetByName(SHEET.COMPANIES);
  const sourceData   = sourceSheet.getDataRange().getValues();
  const sourceHdrs   = sourceData[0].map(String);
  const targetData   = targetSheet.getDataRange().getValues();
  const targetHdrs   = targetData[0].map(String);
  const targetIdCol  = targetHdrs.indexOf('companyId');
  const existingIds  = new Set(targetData.slice(1).map(r => String(r[targetIdCol])));

  const sg = (row, name) => { const i = sourceHdrs.indexOf(name); return i >= 0 ? row[i] : ''; };

  let imported = 0;
  const baseOrder = targetData.length; // ヘッダー込み行数 = 既存企業数+1
  sourceData.slice(1).forEach(row => {
    const companyId = String(sg(row, 'companyId'));
    if (!companyId || existingIds.has(companyId)) return;
    const newRow = targetHdrs.map(h => {
      switch (h) {
        case 'companyId':  return companyId;
        case '企業名':     return String(sg(row, '企業名'));
        case 'stampKey':   return String(sg(row, 'stampKey')  || '');
        case 'viewKey':    return String(sg(row, 'viewKey')   || '');
        case '表示順':     return baseOrder + imported;
        case 'hasNfc':     return String(sg(row, 'hasNfc')    || 'false');
        case 'logoUrl':    return String(sg(row, 'logoUrl')   || '');
        case 'stampRally': return String(sg(row, 'stampRally')|| 'true');
        default:           return '';
      }
    });
    targetSheet.appendRow(newRow);
    existingIds.add(companyId);
    imported++;
  });

  return ok_({ imported });
}

// ============================================================
// アクション: submitCompanyEntry（企業ブース出展申込・公開POST）
// ============================================================

function actionSubmitCompanyEntry_(p) {
  const required = ['companyName','companyShort','director','contact','tel','contactTel','email','zip','prefecture','address','content','booth','demo'];
  for (const f of required) if (!p[f] && p[f] !== 0) return err_('missing_params');

  // イベント解決: パラメータ優先、なければ当日のアクティブイベント
  let eventId = p.event;
  if (eventId) {
    // 明示指定の場合も状態チェックを必ず行う（直接POSTによるバイパス対策）
    if (isEventInactive_(eventId)) return err_('event_inactive');
  } else {
    const evSheet = masterSs_().getSheetByName(SHEET.EVENT_LIST);
    if (!evSheet || evSheet.getLastRow() < 2) return err_('no_active_event');
    const evData    = evSheet.getDataRange().getValues();
    const evHeaders = evData[0].map(String);
    const idCol     = evHeaders.indexOf('eventId');
    const startCol  = evHeaders.indexOf('開始日');
    const endCol    = evHeaders.indexOf('終了日');
    const statCol   = evHeaders.indexOf('状態');
    const today = new Date(); today.setHours(0,0,0,0);
    const evRow = evData.slice(1).find(r => {
      if (statCol >= 0 && (r[statCol] === '公開停止' || r[statCol] === '完了')) return false;
      const s = new Date(r[startCol]); s.setHours(0,0,0,0);
      const e = new Date(r[endCol]);   e.setHours(23,59,59,999);
      return today >= s && today <= e;
    });
    if (!evRow) return err_('no_active_event');
    eventId = String(evRow[idCol]);
  }

  const ss    = eventSs_(eventId);
  let   sheet = ss.getSheetByName(SHEET.COMPANY_ENTRIES);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET.COMPANY_ENTRIES);
    sheet.appendRow([
      '申込日時','eventId',
      '企業名正式','社名略称','代表者名','担当者名','電話番号','担当者電話','メールアドレス',
      '郵便番号','都道府県','住所','部署名',
      '出展内容','ブース区画','展示車両数','デモ走行','デモ走行詳細',
      '人パス','車両パス','昼食土','昼食日',
      '備考','状態',
    ]);
    sheet.setFrozenRows(1);
  }

  const toInt = v => { const n = parseInt(v, 10); return isNaN(n) ? 0 : n; };
  // 負数はフォームの min=0 では防げない（直接POSTで素通りする）。集計が壊れるので弾く。
  // 上限は「大口の申込を誤って拒否しない」ため意図的に見ない（運用判断）。
  const NUM_FIELDS = { carCount:'展示車両数', personPass:'人パス', carPass:'車両パス',
                       lunchSat:'昼食土', lunchSun:'昼食日' };
  for (const k in NUM_FIELDS) {
    if (toInt(p[k]) < 0) return err_('negative_' + k);
  }

  const rowValues = {
    '企業名正式': p.companyName, '社名略称': p.companyShort || '',
    '代表者名': p.director, '担当者名': p.contact,
    '電話番号': p.tel, '担当者電話': p.contactTel, 'メールアドレス': p.email,
    '郵便番号': p.zip, '都道府県': p.prefecture || '', '住所': p.address,
    '部署名': p.department || '',
    '出展内容': p.content, 'ブース区画': p.booth, '展示車両数': toInt(p.carCount),
    'デモ走行': p.demo, 'デモ走行詳細': p.demoDetail || '',
    '人パス': toInt(p.personPass), '車両パス': toInt(p.carPass),
    '昼食土': toInt(p.lunchSat), '昼食日': toInt(p.lunchSun),
    '備考': p.note || '', '状態': '受付済',
  };

  // 同時申込での二重行を防ぐため、既存行の探索〜書き込みをロック内で完結させる
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (e2) { return err_('lock_timeout'); }
  try {
    const data    = sheet.getDataRange().getValues();
    const headers = data[0].map(String);
    const nameCol = headers.indexOf('企業名正式');
    const mailCol = headers.indexOf('メールアドレス');

    // 部署名列は途中で追加したフィールドのため、この機能追加より前に作られたシートには
    // まだ存在しない。無ければ末尾に自動追加する（手動でのシート編集を不要にする）。
    if (headers.indexOf('部署名') < 0) {
      sheet.getRange(1, headers.length + 1).setValue('部署名');
      headers.push('部署名');
    }

    // 同一企業の判定: 企業名正式 または メールアドレス のどちらかが一致したら上書き
    // （通信再送・二重申込で別行が増え、食券や人パスが二重計上されるのを防ぐ）
    const normName = String(p.companyName || '').trim();
    const normMail = String(p.email || '').trim().toLowerCase();
    let rowIdx = -1;
    for (let r = 1; r < data.length; r++) {
      const rn = nameCol >= 0 ? String(data[r][nameCol] || '').trim() : '';
      const rm = mailCol >= 0 ? String(data[r][mailCol] || '').trim().toLowerCase() : '';
      if ((normName && rn === normName) || (normMail && rm === normMail)) { rowIdx = r; break; }
    }

    if (rowIdx < 0) {
      // 新規受付（従来どおり末尾へ追加）
      const rowData = headers.map(h => {
        if (h === '申込日時') return new Date();
        if (h === 'eventId')  return eventId;
        return rowValues[h] !== undefined ? rowValues[h] : '';
      });
      const newRow = sheet.getLastRow() + 1;
      // 電話番号・担当者電話・郵便番号はテキスト書式に（先頭ゼロ保持。書き込み前に設定）
      ['電話番号', '担当者電話', '郵便番号'].forEach(h => {
        const c = headers.indexOf(h);
        if (c >= 0) sheet.getRange(newRow, c + 1).setNumberFormat('@');
      });
      sheet.getRange(newRow, 1, 1, rowData.length).setValues([rowData]);
      return ok_({ message: '受付しました', updated: false });
    }

    // 既存企業の再申込 → 上書きし、変わった項目だけを変更履歴へ残す
    const changes = [];
    Object.keys(rowValues).forEach(h => {
      const c = headers.indexOf(h);
      if (c < 0) return;
      const oldVal = data[rowIdx][c];
      const newVal = rowValues[h];
      if (String(oldVal) === String(newVal)) return;
      if (h !== '状態') changes.push(h + ': ' + oldVal + ' → ' + newVal);
      sheet.getRange(rowIdx + 1, c + 1).setValue(newVal);
    });
    const tsCol = headers.indexOf('申込日時');
    if (tsCol >= 0) sheet.getRange(rowIdx + 1, tsCol + 1).setValue(new Date());

    appendEntryChangeLog_(sheet, headers, rowIdx, '企業からの再申込', changes);
    return ok_({ message: '受付しました（既存の申込を更新しました）', updated: true });
  } finally {
    lock.releaseLock();
  }
}

// 出展申込の「変更履歴」列へ1行追記する。列が無ければ末尾に作る。
// 企業からの再申込と事務局の手直しを同じ欄に、区別できる形で残す。
// 表示は管理画面の申込詳細モーダル最下部（js/admin.js の '変更履歴' グループ）。
function appendEntryChangeLog_(sheet, headers, rowIdx, source, changes) {
  if (!changes || !changes.length) return;
  let logCol = headers.indexOf('変更履歴');
  if (logCol < 0) {
    logCol = headers.length;
    sheet.getRange(1, logCol + 1).setValue('変更履歴');
  }
  const tz = Session.getScriptTimeZone();
  const ts = Utilities.formatDate(new Date(), tz, 'MM/dd HH:mm');
  const prev  = String(sheet.getRange(rowIdx + 1, logCol + 1).getValue() || '');
  const entry = '[' + ts + '] ' + source + ' — ' + changes.join(', ');
  sheet.getRange(rowIdx + 1, logCol + 1).setValue(prev ? prev + '\n' + entry : entry);
}

function actionRegisterSchoolEntry_(p) {

  for (const k of required) {
    if (!p[k] && p[k] !== 0) return err_('missing_' + k);
  }
  // ⚠ 承諾書はフロント(register-school.js)で必須にしているが、サーバー側でも必ず検証する。
  //   フォームを経由しない送信や、フロントの検証が抜けた場合に、承諾書の無い申込が
  //   「受付しました」として記録されるのを防ぐ。
  if (!p.approval || !p.approval.b64) return err_('missing_approval');

  const ss = eventSs_(p.event);
  let sheet = ss.getSheetByName(SHEET.SCHOOL_ENTRIES);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET.SCHOOL_ENTRIES);
    sheet.getRange(1, 1, 1, SCHOOL_ENTRY_COLS.length).setValues([SCHOOL_ENTRY_COLS]);
    sheet.setFrozenRows(1);
  }

  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (e) { return err_('lock_timeout'); }
  try {
    const data    = sheet.getDataRange().getValues();
    const headers = data[0].map(String);
    const idx     = n => headers.indexOf(n);
    const nameCol = idx('学校名');

    // 承諾書ファイルを保存
    // ⚠ 以前は例外を握り潰して approvalUrl を空のまま先へ進めていたため、Driveの一時障害で
    //   保存に失敗しても申込は「受付しました」となり、確認メールまで送られていた。
    //   事務局は承諾書が無いことに気づけない。不完全な申込を残すより、エラーを返して
    //   申込者に再送信してもらう方が安全（2026-08-17 ユーザー判断）。
    let approvalUrl = '';
    try {
      const folder = getSchoolEntryEventFolder_(p.event);
      const bytes  = Utilities.base64Decode(p.approval.b64);
      const fname  = [Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss'),
                      String(p.schoolName).trim(), (p.approval.name || 'approval')].join('_');
      const blob   = Utilities.newBlob(bytes, p.approval.mime || 'application/octet-stream', fname);
      approvalUrl  = folder.createFile(blob).getUrl();
    } catch (err) {
      console.error('school entry file save', err);
      return err_('approval_save_failed');
    }
    if (!approvalUrl) return err_('approval_save_failed');

    const now = new Date();
    const normSchool = String(p.schoolName).trim();
    const existingRowIdx = data.slice(1).findIndex(r => String(r[nameCol]).trim() === normSchool);
    let isUpdate = false;
    let updateCount = 1;
    let historyStr = '';
    if (existingRowIdx >= 0) {
      isUpdate = true;
      const existRow = data[existingRowIdx + 1];
      const prevUrl  = String(existRow[idx('承諾書URL')] || '');
      const prevHist = String(existRow[idx('承諾書履歴')] || '');
      updateCount    = (Number(existRow[idx('更新回数')]) || 1) + 1;
      if (prevUrl) {
        const stamp = Utilities.formatDate(new Date(existRow[idx('提出日時')] || now), Session.getScriptTimeZone(), 'yyyy/MM/dd HH:mm');
        historyStr = (prevHist ? prevHist + '\n' : '') + `[${stamp}] ${prevUrl}`;
      } else {
        historyStr = prevHist;
      }
    }

    const finalApprovalUrl = approvalUrl || (isUpdate ? String(data[existingRowIdx + 1][idx('承諾書URL')] || '') : '');

    const rowMap = {
      '提出日時':       now,
      'eventId':        p.event,
      '学校名':          normSchool,
      '代表者氏名':      String(p.repName || '').trim(),
      '代表者電話':      String(p.repPhone || ''),
      '代表者メール':    String(p.repEmail || '').trim(),
      '発送先_名義':     String(p.shipName || '').trim(),
      '発送先_郵便番号': String(p.shipPostal || ''),
      '発送先_住所':     String(p.shipAddress || '').trim(),
      '発送先_電話':     String(p.shipPhone || ''),
      '車両入場証枚数':  String(p.carPassCount || ''),
      '学校許可取得':    String(p.schoolPermission || ''),
      '事務局への連絡事項': String(p.note || ''),
      '承諾書URL':       finalApprovalUrl,
      '承諾書履歴':      historyStr,
      '規則同意':        p.ruleConsent === 'true' || p.ruleConsent === true ? 'true' : 'false',
      '更新回数':        updateCount,
    };
    const newRow = SCHOOL_ENTRY_COLS.map(h => rowMap[h] !== undefined ? rowMap[h] : '');

    const targetRow = isUpdate ? existingRowIdx + 2 : sheet.getLastRow() + 1;
    // 電話・郵便番号をテキスト書式に（書き込み前）
    ['代表者電話','発送先_郵便番号','発送先_電話'].forEach(col => {
      const ci = SCHOOL_ENTRY_COLS.indexOf(col);
      if (ci >= 0) sheet.getRange(targetRow, ci + 1).setNumberFormat('@');
    });
    sheet.getRange(targetRow, 1, 1, newRow.length).setValues([newRow]);

    // 確認メール送信（非同期・失敗しても登録は成立）
    sendSchoolEntryConfirmMail_(p.event, {
      to:       p.repEmail,
      repName:  p.repName,
      school:   normSchool,
      isUpdate: isUpdate,
    });

    return ok_({ message: isUpdate ? '内容を更新しました' : '受付しました', isUpdate: isUpdate });
  } finally {
    lock.releaseLock();
  }
}

/** 出場校エントリー確認メール
 *  件名・本文は設定シート（schoolEntryMailSubject / schoolEntryMailSubjectUpdate / schoolEntryMailBody）で編集可。
 *  差し込み: {repName} 代表者氏名, {school} 学校名, {eventName} 大会名, {updateNote} 更新時の追記文
 */
function sendSchoolEntryConfirmMail_(eventId, info) {
  if (!info.to) return;
  try {
    const cfg    = getConfig_(eventId);
    const evName = cfg.eventName || eventId;
    const subjNewTpl = cfg.schoolEntryMailSubject       || '【{eventName}】出場校エントリーを受け付けました';
    const subjUpdTpl = cfg.schoolEntryMailSubjectUpdate || '【{eventName}】出場校エントリーの更新を受け付けました';
    const bodyTpl    = cfg.schoolEntryMailBody ||
      ('{repName} 様\n\n' +
       '{eventName} の出場校エントリー（{school}）を受け付けました。\n\n' +
       '{updateNote}' +
       '入場パスは後日発送いたします。\n\n' +
       '── FORMULA GYMKHANA 事務局');
    const updateNote = info.isUpdate
      ? '以前の入力内容は破棄され、今回の内容が最新として保存されました。\n\n'
      : '内容に変更がある場合は、同じフォームから再度送信してください（最新のもので上書きされます）。\n\n';
    const fill = s => String(s)
      .replace(/\{repName\}/g,   info.repName || '')
      .replace(/\{school\}/g,    info.school  || '')
      .replace(/\{eventName\}/g, evName)
      .replace(/\{updateNote\}/g, updateNote);
    const subject = fill(info.isUpdate ? subjUpdTpl : subjNewTpl);
    const body    = fill(bodyTpl);
    MailApp.sendEmail({ to: String(info.to), subject: subject, body: body });
  } catch (err) {
    console.error('sendSchoolEntryConfirmMail_ error:', err);
  }
}

// アクション: adminGetSchoolEntries（管理者向け出場校エントリー一覧）
// ============================================================

function actionAdminGetSchoolEntries_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event) return err_('missing_event');

  const sheet = eventSs_(p.event).getSheetByName(SHEET.SCHOOL_ENTRIES);
  if (!sheet || sheet.getLastRow() < 2) return ok_({ entries: [] });

  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);
  const tz      = Session.getScriptTimeZone();

  const entries = data.slice(1).map(row => {
    const o = {};
    headers.forEach((h, i) => {
      const v = row[i];
      if (v instanceof Date) o[h] = Utilities.formatDate(v, tz, 'yyyy/MM/dd HH:mm');
      else o[h] = v == null ? '' : String(v);
    });
    return o;
  });
  return ok_({ entries: entries });
}

// ============================================================
// アクション: adminUpdateEntry（出展申込の部分更新）
// ============================================================
function actionAdminUpdateEntry_(p) {
  const e = validateAdmin_(p);
  if (e) return e;
  if (!p.event)   return err_('missing_event');
  if (!p.company) return err_('missing_params');
  if (!p.updates || typeof p.updates !== 'object') return err_('missing_params');

  const sheet = eventSs_(p.event).getSheetByName(SHEET.COMPANY_ENTRIES);
  if (!sheet || sheet.getLastRow() < 2) return err_('not_found');

  const data    = sheet.getDataRange().getValues();
  const headers = data[0].map(String);

  var rowIdx = -1;
  var nameCol = headers.indexOf('企業名正式');
  var abbCol  = headers.indexOf('社名略称');
  for (var r = 1; r < data.length; r++) {
    if (String(data[r][nameCol]) === p.company || String(data[r][abbCol]) === p.company) {
      rowIdx = r;
      break;
    }
  }
  if (rowIdx < 0) return err_('not_found');

  var allowed = ['展示車両数','人パス','車両パス','昼食土','昼食日','デモ走行','部署名'];
  var changes = [];
  for (var key in p.updates) {
    if (allowed.indexOf(key) < 0) continue;
    var col = headers.indexOf(key);
    if (col < 0) continue;
    var oldVal = data[rowIdx][col];
    var newVal = p.updates[key];
    if (String(oldVal) !== String(newVal)) {
      changes.push(key + ': ' + oldVal + ' → ' + newVal);
      sheet.getRange(rowIdx + 1, col + 1).setValue(newVal);
    }
  }

  // 企業からの再申込と同じ欄へ、どちらが変えたのか分かる形で残す
  appendEntryChangeLog_(sheet, headers, rowIdx, '事務局の修正', changes);
  return ok_({ updated: true });
}
